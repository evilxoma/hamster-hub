local function ensureExecutorFolders()
	if typeof(makefolder) ~= "function" then
		return false
	end

local function executorFileExists(path)
	if typeof(isfile) ~= "function" then
		return false
	end

	local success, result = pcall(isfile, path)
	return success and result == true
end

local function executorWrite(path, content)
	if typeof(writefile) ~= "function" then
		return false, "writefile is unavailable"
	end

	ensureExecutorFolders()
	local success, err = pcall(writefile, path, content)
	return success, err
end

local function executorRead(path)
	if typeof(readfile) ~= "function" or not executorFileExists(path) then
		return nil
	end

	local success, content = pcall(readfile, path)
	return success and content or nil
end

local function loadExternalSettings()
	local externalSettings = executorRead(SETTINGS_PATH)

	if not externalSettings then
		return false
	end

	local settingsOk, settingsData = pcall(function()
		return HttpService:JSONDecode(externalSettings)
	end)

	if not settingsOk or typeof(settingsData) ~= "table" then
		return false
	end

	if settingsData.AutoLoad ~= nil then
		configAutoloadEnabled = settingsData.AutoLoad == true
	end

	if not configAutoloadEnabled then
		return true
	end

	for _, key in ipairs({
		"AutoTeleport",
		"AutoReplay",
		"AutoBackToLobby",
		"AutoVipAfterGame",
		"AutoSkip",
		"AutoX2Speed",
	}) do
		if settingsData[key] ~= nil then
			profileSnapshot.Settings[key] =
				settingsData[key] == true
		end
	end

	return true
end

local function loadSavedSnapshot()
	local success, savedValue = pcall(function()
		return TeleportService:GetTeleportSetting(SNAPSHOT_KEY)
	end)

	if not success or typeof(savedValue) ~= "string" or savedValue == "" then
		return
	end

	local decodedSuccess, decoded = pcall(function()
		return HttpService:JSONDecode(savedValue)
	end)

	if not decodedSuccess or typeof(decoded) ~= "table" then
		return
	end

	if decoded.Coins ~= nil then
		profileSnapshot.Coins = tostring(decoded.Coins)
	end

	if decoded.Gems ~= nil then
		profileSnapshot.Gems = tostring(decoded.Gems)
	end

	if typeof(decoded.Settings) == "table" then
		profileSnapshot.Settings.AutoTeleport =
			decoded.Settings.AutoTeleport ~= false
		profileSnapshot.Settings.AutoReplay =
			decoded.Settings.AutoReplay == true

		profileSnapshot.Settings.AutoBackToLobby =
			decoded.Settings.AutoBackToLobby == true

		profileSnapshot.Settings.AutoVipAfterGame =
			decoded.Settings.AutoVipAfterGame == true

		profileSnapshot.Settings.AutoSkip =
			decoded.Settings.AutoSkip == true

		profileSnapshot.Settings.AutoX2Speed =
			decoded.Settings.AutoX2Speed == true

		if table.find(
			ACTION_GROUP_ORDER,
			tostring(decoded.Settings.CurrentActionGroup)
		) then
			profileSnapshot.Settings.CurrentActionGroup =
				tostring(decoded.Settings.CurrentActionGroup)
		end

		if typeof(decoded.Settings.EnabledGroups) == "table" then
			for _, groupName in ipairs(ACTION_GROUP_ORDER) do
				local storedValue =
					decoded.Settings.EnabledGroups[groupName]

				if storedValue ~= nil then
					profileSnapshot.Settings.EnabledGroups[groupName] =
						storedValue == true
				end
			end
		end

		if profileSnapshot.Settings.AutoVipAfterGame then
			profileSnapshot.Settings.AutoReplay = false
			profileSnapshot.Settings.AutoBackToLobby = false
		elseif profileSnapshot.Settings.AutoReplay then
			profileSnapshot.Settings.AutoBackToLobby = false
		end
	end

	if typeof(decoded.Deck) == "table" then
		for slotNumber = 1, 6 do
			local savedSlot = decoded.Deck[slotNumber]
				or decoded.Deck[tostring(slotNumber)]

			if typeof(savedSlot) == "table" then
				profileSnapshot.Deck[slotNumber] = {
					Slot = slotNumber,
					Image = tostring(savedSlot.Image or ""),
					Name = tostring(savedSlot.Name or ""),
				}
			end
		end
	end

	savedSnapshotLoaded = true
end

local function appendFailureReport(message, context)
	if typeof(writefile) ~= "function" and typeof(appendfile) ~= "function" then
		return false
	end
	context = typeof(context) == "table" and context or {}
	local info = workspace:FindFirstChild("Info")
	local function value(name)
		local object = info and info:FindFirstChild(name)
		return object and object:IsA("ValueBase") and object.Value or "Unknown"
	end
	local report = table.concat({
		"",
		"==================================================",
		"UTC: " .. os.date("!%Y-%m-%dT%H:%M:%SZ"),
		"Stage: " .. tostring(context.Stage or "Hamster Hub"),
		"Message: " .. tostring(message),
		"Map: " .. tostring(value("Map")),
		"Mode: " .. tostring(value("Mode")),
		"Wave: " .. tostring(value("Wave")),
		"==================================================",
	}, "\n") .. "\n"
	local ok = pcall(function()
		if typeof(appendfile) == "function" then
			appendfile(ERROR_LOG_PATH, report)
		else
			local existing = executorRead(ERROR_LOG_PATH) or ""
			executorWrite(ERROR_LOG_PATH, existing .. report)
		end
	end)
	if ok then
		updateRayfieldLabel("Failure", "Last failure saved to " .. ERROR_LOG_PATH)
	end
	return ok
end

local function findDeckSlotsContainer()
	local mainGui = playerGui:FindFirstChild("Main")

	if mainGui then
		local mainFrameObject = mainGui:FindFirstChild("Main")

		if mainFrameObject then
			local slotsOuter = mainFrameObject:FindFirstChild("Slots")
			local slotsInner = slotsOuter
				and slotsOuter:FindFirstChild("Slots")

			if slotsInner and slotsInner:FindFirstChild("Slot1") then
				return slotsInner
			end
		end
	end

	local gameGui = playerGui:FindFirstChild("GameGui")

	if gameGui then
		local slotsOuter = gameGui:FindFirstChild("Slots")
		local slotsInner = slotsOuter
			and slotsOuter:FindFirstChild("Slots")

		if slotsInner and slotsInner:FindFirstChild("Slot1") then
			return slotsInner
		end
	end

	for _, object in ipairs(playerGui:GetDescendants()) do
		if object.Name == "Slots"
			and object:FindFirstChild("Slot1")
			and object:FindFirstChild("Slot2") then
			return object
		end
	end

	return nil
end

local function getSlotImage(slotObject)
	if slotObject:IsA("ImageButton")
		or slotObject:IsA("ImageLabel") then
		if slotObject.Image and slotObject.Image ~= "" then
			return slotObject.Image
		end
	end

	local iconImage = slotObject:FindFirstChild("IconImage", true)

	if iconImage
		and (iconImage:IsA("ImageButton")
			or iconImage:IsA("ImageLabel"))
		and iconImage.Image ~= "" then
		return iconImage.Image
	end

	for _, descendant in ipairs(slotObject:GetDescendants()) do
		if descendant:IsA("ImageButton")
			or descendant:IsA("ImageLabel") then
			if descendant.Image ~= ""
				and not string.find(
					string.lower(descendant.Name),
					"shiny"
				) then
				return descendant.Image
			end
		end
	end

	return ""
end

local function getSlotUnitName(slotObject, image)
	local attributeNames = {
		"TowerName",
		"UnitName",
		"Name",
	}

	for _, attributeName in ipairs(attributeNames) do
		local value = slotObject:GetAttribute(attributeName)

		if typeof(value) == "string"
			and value ~= ""
			and value ~= slotObject.Name then
			return value
		end
	end

	local mappedName = towerNamesByImage[normalizeImage(image)]

	if mappedName then
		return mappedName
	end

	for _, descendant in ipairs(slotObject:GetDescendants()) do
		if descendant:IsA("StringValue") then
			local lowerName = string.lower(descendant.Name)

			if string.find(lowerName, "tower")
				or string.find(lowerName, "unit") then
				if descendant.Value ~= "" then
					return descendant.Value
				end
			end
		end
	end

	return ""
end

local function captureDeck()
	local slotsContainer = findDeckSlotsContainer()

	if not slotsContainer then
		return false
	end

	local foundAnyUnit = false

	for slotNumber = 1, 6 do
		local slotObject = slotsContainer:FindFirstChild(
			"Slot" .. slotNumber
		)

		if slotObject then
			local image = getSlotImage(slotObject)
			local unitName = getSlotUnitName(slotObject, image)

			profileSnapshot.Deck[slotNumber] = {
				Slot = slotNumber,
				Image = image,
				Name = unitName,
			}

			if image ~= "" or unitName ~= "" then
				foundAnyUnit = true
			end
		end
	end

	return foundAnyUnit
end

local function roundNumber(value, digits)
	local multiplier = 10 ^ (digits or 0)
	return math.round(value * multiplier) / multiplier
end

local function getValue(containerObject, childName)
	local valueObject = containerObject
		and containerObject:FindFirstChild(childName)

	if valueObject and valueObject:IsA("ValueBase") then
		return valueObject.Value
	end

	return nil
end

local function getTowerCFrame(tower)
	if not tower or not tower:IsA("Model") then
		return nil
	end

	local success, pivot = pcall(function()
		return tower:GetPivot()
	end)

	if success then
		return pivot
	end

	local root = tower.PrimaryPart
		or tower:FindFirstChild("HumanoidRootPart")
		or tower:FindFirstChildWhichIsA("BasePart", true)

	return root and root.CFrame or nil
end

local function isOurTower(config)
	local owner = getValue(config, "Owner")

	if owner == nil or tostring(owner) == "" then
		return true
	end

	if typeof(owner) == "Instance" and owner:IsA("Player") then
		return owner == player
	end

	return tostring(owner or "") == player.Name
end

local function getOrAssignTowerId(tower, preferredId)
	if not tower then
		return nil
	end

	local existingId = towerIds[tower]

	if existingId then
		return existingId
	end

	local unitId = tonumber(preferredId)

	if not unitId then
		unitId = nextTowerId
		nextTowerId = nextTowerId + 1
	else
		nextTowerId = math.max(nextTowerId, unitId + 1)
	end

	towerIds[tower] = unitId
	return unitId
end

local function getTowerSnapshot(tower, deferUnitId)
	if not tower or not tower:IsA("Model") then
		return nil
	end

	local config = tower:FindFirstChild("Config")

	if not config then
		config = tower:WaitForChild("Config", 2)
	end

	if not config or not isOurTower(config) then
		return nil
	end

	local towerCFrame = getTowerCFrame(tower)

	if not towerCFrame then
		return nil
	end

	local towerName = getValue(config, "TowerName")
		or getValue(config, "UnitName")
		or tower.Name

	local level = tonumber(getValue(config, "Level")) or 1
	local currentPrice =
		tonumber(getValue(config, "Price")) or 0

	local nextPrice = nil
	local upgradeObject = config:FindFirstChild("Upgrade")

	if upgradeObject and upgradeObject:IsA("ObjectValue") then
		local nextTower = upgradeObject.Value
		local nextConfig =
			nextTower and nextTower:FindFirstChild("Config")

		nextPrice = tonumber(getValue(nextConfig, "Price"))
	end

	local components = {towerCFrame:GetComponents()}

	for index, component in ipairs(components) do
		components[index] = roundNumber(component, 6)
	end

	local captureInfo = workspace:FindFirstChild("Info")
	local capturedTiming = getValue(captureInfo, "Time")

	return {
		Model = tower,
		UnitId = deferUnitId
			and towerIds[tower]
			or (
				not deferUnitId
				and getOrAssignTowerId(tower)
				or nil
			),
		Name = tostring(towerName),
		Template = tower.Name,
		Level = level,
		Price = currentPrice,
		SellValue = math.floor(currentPrice / 2),
		NextPrice = nextPrice,
		TargetMode = tostring(
			getValue(config, "TargetMode") or "First"
		),
		Position = towerCFrame.Position,
		CFrame = components,
		Wave = tonumber(getValue(captureInfo, "Wave")) or 0,
		Timing = capturedTiming ~= nil
			and tostring(capturedTiming)
			or "",
		CapturedAt = os.clock(),
	}
end

local function formatCoordinates(position)
	return string.format(
		"(%.2f, %.2f, %.2f)",
		position.X,
		position.Y,
		position.Z
	)
end

local function formatTiming(seconds)
	local safeSeconds = math.max(0, tonumber(seconds) or 0)
	local minutes = math.floor(safeSeconds / 60)
	local remainingSeconds = safeSeconds - minutes * 60

	return string.format(
		"%02d:%05.2f",
		minutes,
		remainingSeconds
	)
end

local function getCurrentTiming(fallbackSeconds)
	local info = workspace:FindFirstChild("Info")
	local gameTime = getValue(info, "Time")

	if gameTime ~= nil and tostring(gameTime) ~= "" then
		return tostring(gameTime)
	end

	return formatTiming(fallbackSeconds)
end

	local function applySavedLoadout()
		local loadout = STRATEGY.Loadout or STRATEGY.Deck
		if type(loadout) ~= "table" then
			return false, "Strategy Loadout is missing."
		end

		local events = ReplicatedStorage:FindFirstChild("Events")
		local functions = ReplicatedStorage:FindFirstChild("Functions")
		local remote
		for _, container in ipairs({events, functions}) do
			if container then
				for _, name in ipairs({
					"LoadLoadout",
					"SetLoadout",
					"EquipLoadout",
					"ApplyLoadout",
					"ChangeLoadout",
				}) do
					local candidate = container:FindFirstChild(name, true)
					if candidate
						and (candidate:IsA("RemoteEvent")
							or candidate:IsA("RemoteFunction")) then
						remote = candidate
						break
					end
				end
			end
			if remote then
				break
			end
		end

		if remote then
			local sent = pcall(function()
				if remote:IsA("RemoteEvent") then
					remote:FireServer(loadout)
				else
					remote:InvokeServer(loadout)
				end
			end)
			if not sent then
				return false, "The saved Loadout could not be applied."
			end
			log("Saved Loadout applied.")
		else
			log("No loadout remote was exposed; keeping the current Loadout.")
		end
		return true
	end

	if lobbyPresent then
		if STRATEGY.AutoEnterMatch == false then
			fail("Auto match entry is disabled in the strategy.")
			environment.HamsterStrategyPlayerRunning = false
			return
		end

		local loadoutApplied, loadoutError = applySavedLoadout()
		if not loadoutApplied then
			fail(loadoutError, nil, nil, "Loadout")
			environment.HamsterStrategyPlayerRunning = false
			return
		end

		local success, lobbyError = runLobbyAutomation()

		if not success then
			fail(lobbyError)
		end

		environment.HamsterStrategyPlayerRunning = false
		return
	end

	if STRATEGY.PlaceId
		and STRATEGY.PlaceId ~= 0
		and game.PlaceId ~= STRATEGY.PlaceId then
		fail(
			"PlaceId does not match the strategy: required "
				.. tostring(STRATEGY.PlaceId)
				.. ", current "
				.. tostring(game.PlaceId)
				.. "."
		)
		environment.HamsterStrategyPlayerRunning = false
		return
	end

	local info = workspace:WaitForChild("Info", 45)
	local functions =
		ReplicatedStorage:WaitForChild("Functions", 45)

	local events =
		ReplicatedStorage:WaitForChild("Events", 45)

	if not info or not functions or not events then
		fail("Required Info, Functions, or Events folders were not found.")
		environment.HamsterStrategyPlayerRunning = false
		return
	end

	local SpawnTower = functions:FindFirstChild("SpawnTower")
	local SellTower = functions:FindFirstChild("SellTower")

	local ChangeTowerMode =
		functions:FindFirstChild("ChangeTowerMode")

	local GetData = functions:FindFirstChild("GetData")
	local VoteSkip = events:FindFirstChild("VoteSkip")
	local Speed = events:FindFirstChild("Speed")

	if not SpawnTower
		or not SellTower
		or not ChangeTowerMode
		or not GetData then
		fail("Unit control RemoteFunctions were not found.")
		environment.HamsterStrategyPlayerRunning = false
		return
	end

	local MAX_RETRIES = 3
	local RETRY_DELAY = 0.35
	local ACTION_COOLDOWN = 0.1
	local POSITION_TOLERANCE = 1.5
	local SPAWN_CONFIRM_TIMEOUT = 3.5
	local SUPPORTED_FORMAT_VERSION = 7
	local towerRegistry = {}
	local towerUnitIds = {}

	local function readValue(container, name)
		local valueObject =
			container and container:FindFirstChild(name)

		if valueObject and valueObject:IsA("ValueBase") then
			return valueObject.Value
		end

		return nil
	end

	local function toNumber(value)
		local numericValue = tonumber(value)

		if numericValue then
			return numericValue
		end

		local textValue = tostring(value or "")
		local firstNumber = string.match(
			textValue,
			"[-+]?%d+%.?%d*"
		)

		return tonumber(firstNumber)
	end

	local function isGameActive()
		local running = readValue(info, "GameRunning")

		if running == true
			or running == 1
			or tostring(running):lower() == "true"
			or tostring(running):lower() == "running" then
			return true
		end

		local waveNumber =
			toNumber(readValue(info, "Wave")) or 0

		return waveNumber > 0
	end

	local function currentWave()
		return toNumber(readValue(info, "Wave")) or 0
	end

	local function normalizedChoice(value)
		local textValue = tostring(value or "")

		if textValue == ""
			or textValue == "Unknown"
			or textValue == "Any" then
			return nil
		end

		return textValue
	end

	local function findSelectionRemote(names)
		for _, remoteName in ipairs(names) do
			local remote = events:FindFirstChild(remoteName, true)

			if remote
				and (remote:IsA("RemoteEvent")
					or remote:IsA("RemoteFunction")) then
				return remote
			end
		end

		return nil
	end

	local function sendSelection(remote, value)
		if not remote or not value then
			return false
		end

		local success = pcall(function()
			if remote:IsA("RemoteEvent") then
				remote:FireServer(value)
			else
				remote:InvokeServer(value)
			end
		end)

		return success
	end

	local function startMatchSelection()
		local desiredMode = normalizedChoice(STRATEGY.Mode)

		local modeRemote = findSelectionRemote({
			"VoteForMap",
			"VoteForMode",
			"SelectMode",
			"SetMode",
		})

		if not desiredMode then
			return
		end

		task.spawn(function()
			local lastModeVoteAt = 0

			while info.Parent and not isGameActive() do
				local voting = readValue(info, "Voting") == true

				if voting then
					local currentMode = tostring(
						readValue(info, "Mode") or ""
					)

					if desiredMode
						and currentMode ~= desiredMode
						and os.clock() - lastModeVoteAt >= 1 then
						lastModeVoteAt = os.clock()
						sendSelection(modeRemote, desiredMode)
					end

				end

				task.wait(0.2)
			end
		end)
	end

	local function waitForActiveGame()
		local startedWaitingAt = os.clock()

		while info.Parent
			and not isGameActive()
			and os.clock() - startedWaitingAt < 90 do
			overlayLabel.Text =
				"Strategy Player: waiting for a match..."
			task.wait(0.25)
		end

		if not info.Parent or not isGameActive() then
			fail(
				"Could not detect an active match in Info.GameRunning or Info.Wave.",
				nil,
				nil,
				"Match Detection"
			)
			return false
		end

		overlayLabel.Text =
			"Strategy Player: match detected, validating..."
		return true
	end

	local function waitForRecordingStartWave()
		local expectedWave =
			tonumber(STRATEGY.StartWave) or 0

		while isGameActive()
			and currentWave() < expectedWave do
			task.wait(0.1)
		end

		if not isGameActive() then
			return false
		end

		if expectedWave > 0
			and currentWave() > expectedWave then
			fail(
				"The match is already at wave "
					.. tostring(currentWave())
					.. ", but the strategy starts at "
					.. tostring(expectedWave)
					.. "."
			)
			return false
		end

		return true
	end

	local function checkCompatibility()
		local metadata = typeof(STRATEGY.Metadata) == "table"
			and STRATEGY.Metadata
			or {}

		local formatVersion = tonumber(
			metadata.FormatVersion
				or STRATEGY.FormatVersion
				or 1
		) or 1

		if formatVersion > SUPPORTED_FORMAT_VERSION then
			return false,
				"Strategy format "
					.. tostring(formatVersion)
					.. " is newer than supported format "
					.. tostring(SUPPORTED_FORMAT_VERSION)
					.. "."
		end

		local mapName = tostring(
			readValue(info, "Map") or ""
		)

		local modeName = tostring(
			readValue(info, "Mode") or ""
		)

		if STRATEGY.Map
			and STRATEGY.Map ~= "Unknown"
			and mapName ~= STRATEGY.Map then
			return false,
				"Map mismatch. Required "
					.. STRATEGY.Map
					.. ", current "
					.. mapName
		end

		if STRATEGY.Mode
			and STRATEGY.Mode ~= "Unknown"
			and modeName ~= STRATEGY.Mode then
			return false,
				"Mode mismatch. Required "
					.. STRATEGY.Mode
					.. ", current "
					.. modeName
		end

		local success, data = pcall(function()
			return GetData:InvokeServer()
		end)

		if not success or typeof(data) ~= "table" then
			return false,
				"The server did not return the current loadout."
		end

		local deckNames = {}

		for slotNumber = 1, 6 do
			local slot =
				data["Slot" .. tostring(slotNumber)]

			local first = typeof(slot) == "table"
				and slot[1]
				or nil

			local unitName = typeof(first) == "table"
				and (first.name or first.Name)
				or nil

			if unitName then
				deckNames[
					string.lower(tostring(unitName))
				] = true
			end
		end

		for _, action in ipairs(STRATEGY.Actions or {}) do
			if isActionGroupEnabled(action)
				and action.Type == "Place" then
				local requiredName = tostring(
					action.Template or action.Unit or ""
				)

				if requiredName ~= ""
					and not deckNames[
						string.lower(requiredName)
					] then
					return false,
						"The loadout does not contain unit "
							.. requiredName
							.. "."
				end
			end
		end

		if formatVersion >= 4 then
			local placedUnitIds = {}

			for index, action in ipairs(STRATEGY.Actions or {}) do
				local actionType = tostring(action.Type or "")
				local needsUnitId = actionType == "Place"
					or actionType == "Upgrade"
					or actionType == "Sell"
					or actionType == "Target"
				local unitId = tonumber(action.UnitId)

				if needsUnitId and not unitId then
					return false,
						"Action #"
							.. tostring(index)
							.. " is missing UnitId."
				end

				if actionType == "Place" then
					if placedUnitIds[unitId] then
						return false,
							"UnitId #"
								.. tostring(unitId)
								.. " is used by multiple Place actions."
					end

					placedUnitIds[unitId] = true
				elseif needsUnitId and not placedUnitIds[unitId] then
					return false,
						"Action #"
							.. tostring(index)
							.. " references UnitId before its Place action."
				end
			end
		end

		return true
	end

	local function getTowerRoot(tower)
		if not tower or not tower:IsA("Model") then
			return nil
		end

		return tower.PrimaryPart
			or tower:FindFirstChild("HumanoidRootPart")
			or tower:FindFirstChildWhichIsA(
				"BasePart",
				true
			)
	end

	local function isOurTower(tower)
		local config =
			tower and tower:FindFirstChild("Config")

		local owner = readValue(config, "Owner")

		if typeof(owner) == "Instance"
			and owner:IsA("Player") then
			return owner == player
		end

		return tostring(owner or "") == player.Name
	end

	local function actionPosition(action)
		if not action.Position then
			return nil
		end

		return Vector3.new(
			tonumber(action.Position.X) or 0,
			tonumber(action.Position.Y) or 0,
			tonumber(action.Position.Z) or 0
		)
	end

	local function actionCFrame(action)
		if typeof(action.CFrame) == "table"
			and #action.CFrame >= 12 then
			return CFrame.new(table.unpack(action.CFrame))
		end

		local position = actionPosition(action)
		return position and CFrame.new(position) or nil
	end

	local function normalizedName(value)
		return string.lower(tostring(value or ""))
	end

	local function towerMatchesAction(tower, action)
		if not tower then
			return false
		end

		local config = tower:FindFirstChild("Config")
		local towerName = normalizedName(
			readValue(config, "TowerName")
				or readValue(config, "UnitName")
				or tower.Name
		)
		local templateName = normalizedName(tower.Name)
		local expectedNames = {
			action.Unit,
			action.PreviousUnit,
			action.Template,
			action.PreviousTemplate,
		}
		local hasExpectedName = false

		for _, expected in ipairs(expectedNames) do
			local normalizedExpected = normalizedName(expected)

			if normalizedExpected ~= "" then
				hasExpectedName = true

				if towerName == normalizedExpected
					or templateName == normalizedExpected then
					return true
				end
			end
		end

		return not hasExpectedName
	end

	local function getTowerLevel(tower)
		local config = tower and tower:FindFirstChild("Config")
		return tonumber(readValue(config, "Level")) or 1
	end

	local function clearTowerBinding(unitId)
		unitId = tonumber(unitId)

		if not unitId then
			return
		end

		local tower = towerRegistry[unitId]
		towerRegistry[unitId] = nil

		if tower and towerUnitIds[tower] == unitId then
			towerUnitIds[tower] = nil
		end
	end

	local function bindTower(unitId, tower)
		unitId = tonumber(unitId)

		if not unitId or not tower or not tower:IsA("Model") then
			return tower
		end

		local previousTower = towerRegistry[unitId]

		if previousTower and previousTower ~= tower
			and towerUnitIds[previousTower] == unitId then
			towerUnitIds[previousTower] = nil
		end

		local previousUnitId = towerUnitIds[tower]

		if previousUnitId and previousUnitId ~= unitId then
			towerRegistry[previousUnitId] = nil
		end

		towerRegistry[unitId] = tower
		towerUnitIds[tower] = unitId
		return tower
	end

	local function getRegisteredTower(unitId)
		unitId = tonumber(unitId)

		if not unitId then
			return nil
		end

		local tower = towerRegistry[unitId]

		if tower
			and tower.Parent
			and tower:IsA("Model")
			and isOurTower(tower) then
			return tower
		end

		clearTowerBinding(unitId)
		return nil
	end

	local function captureTowerSet()
		local result = {}
		local towers = workspace:FindFirstChild("Towers")

		if towers then
			for _, tower in ipairs(towers:GetChildren()) do
				if tower:IsA("Model") then
					result[tower] = true
				end
			end
		end

		return result
	end

	local function levelIsAllowed(tower, allowedLevels)
		if allowedLevels == nil then
			return true
		end

		local currentLevel = getTowerLevel(tower)

		if typeof(allowedLevels) == "number" then
			return currentLevel == allowedLevels
		end

		if typeof(allowedLevels) == "table" then
			for _, level in ipairs(allowedLevels) do
				if currentLevel == tonumber(level) then
					return true
				end
			end
		end

		return false
	end

	local function findFallbackTower(action, allowedLevels)
		local towers = workspace:FindFirstChild("Towers")
		local position = actionPosition(action)
		local unitId = tonumber(action.UnitId)

		if not towers or not position then
			return nil
		end

		local closest = nil
		local closestDistance = POSITION_TOLERANCE

		for _, tower in ipairs(towers:GetChildren()) do
			if tower:IsA("Model")
				and isOurTower(tower)
				and towerMatchesAction(tower, action)
				and levelIsAllowed(tower, allowedLevels) then
				local boundId = towerUnitIds[tower]

				if not boundId or boundId == unitId then
					local root = getTowerRoot(tower)

					if root then
						local distance =
							(root.Position - position).Magnitude

						if distance <= closestDistance then
							closest = tower
							closestDistance = distance
						end
					end
				end
			end
		end

		if closest and unitId then
			bindTower(unitId, closest)
		end

		return closest
	end

	local function findTower(action, allowedLevels)
		local registered = getRegisteredTower(action.UnitId)

		if registered then
			return registered
		end

		return findFallbackTower(action, allowedLevels)
	end

	local function waitForTowerResult(
		action,
		baseline,
		returnedValue,
		expectedLevel,
		allowExistingTower,
		timeout
	)
		local towers = workspace:FindFirstChild("Towers")

		if not towers then
			return nil
		end

		local returnedTower = typeof(returnedValue) == "Instance"
			and returnedValue:IsA("Model")
			and returnedValue
			or nil
		local expectedPosition = actionPosition(action)
		local unitId = tonumber(action.UnitId)
		local deadline = os.clock() + (timeout or SPAWN_CONFIRM_TIMEOUT)

		repeat
			local candidates = {}

			if returnedTower and returnedTower.Parent then
				table.insert(candidates, returnedTower)
			end

			for _, tower in ipairs(towers:GetChildren()) do
				if tower:IsA("Model") and tower ~= returnedTower then
					table.insert(candidates, tower)
				end
			end

			for _, tower in ipairs(candidates) do
				local isNewTower = not baseline[tower]
				local isAllowedExisting =
					allowExistingTower ~= nil
					and tower == allowExistingTower

				if (isNewTower or isAllowedExisting)
					and tower.Parent
					and isOurTower(tower)
					and towerMatchesAction(tower, action)
					and getTowerLevel(tower) == expectedLevel then
					local boundId = towerUnitIds[tower]
					local root = getTowerRoot(tower)
					local closeEnough = root ~= nil

					if root and expectedPosition then
						closeEnough =
							(root.Position - expectedPosition).Magnitude
							<= POSITION_TOLERANCE
					end

					if closeEnough
						and (not boundId or boundId == unitId) then
						return bindTower(unitId, tower)
					end
				end
			end

			task.wait(0.05)
		until os.clock() >= deadline or not isGameActive()

		return nil
	end

	local function parseAmount(value)
		if typeof(value) == "number" then
			return value
		end

		local textValue = tostring(value or "")
			:gsub("%s", "")
			:gsub(",", "")

		local suffix = string.upper(
			string.match(textValue, "([KMBT])$") or ""
		)

		local numberValue = tonumber(
			string.match(textValue, "[-+]?[%d%.]+") or ""
		)

		if not numberValue then
			return nil
		end

		local multipliers = {
			K = 1e3,
			M = 1e6,
			B = 1e9,
			T = 1e12,
		}

		return numberValue * (multipliers[suffix] or 1)
	end

	local function currentGoldAmount()
		local locations = {
			player,
			player:FindFirstChild("leaderstats"),
			player:FindFirstChild("Stats"),
			player:FindFirstChild("Data"),
		}

		for _, location in ipairs(locations) do
			if location then
				for _, valueName in ipairs({
					"Gold",
					"Cash",
					"Money",
				}) do
					local valueObject = location:FindFirstChild(valueName)

					if valueObject and valueObject:IsA("ValueBase") then
						local amount = parseAmount(valueObject.Value)

						if amount then
							return amount
						end
					end
				end
			end
		end

		local gameGui = player.PlayerGui:FindFirstChild("GameGui")

		if gameGui then
			for _, descendant in ipairs(gameGui:GetDescendants()) do
				if descendant:IsA("TextLabel")
					or descendant:IsA("TextButton") then
					local lowerName = string.lower(descendant.Name)

					if string.find(lowerName, "gold")
						or string.find(lowerName, "cash")
						or string.find(lowerName, "money") then
						local amount = parseAmount(descendant.Text)

						if amount then
							return amount
						end
					end
				end
			end
		end

		return 0
	end

	local function actionRequiredGold(action)
		if STRATEGY.WaitForMoney == false then
			return 0
		end

		if action.Type == "Place"
			or action.Type == "Upgrade" then
			return math.max(0, tonumber(action.Price) or 0)
		end

		return 0
	end

	local function waitForGold(requiredAmount)
		local required = tonumber(requiredAmount) or 0

		if required <= 0 then
			return true
		end

		while isGameActive() do
			if currentGoldAmount() >= required then
				return true
			end

			task.wait(0.1)
		end

		return false
	end

	local function waitForAction(
		action,
		actionIndex,
		lastActionAt
	)
		local expectedWave = tonumber(action.Wave) or 0
		local requiredGold = actionRequiredGold(action)

		while isGameActive() do
			local now = os.clock()
			local currentWaveNumber = currentWave()
			local currentGold = currentGoldAmount()
			local cooldownRemaining = 0

			if lastActionAt then
				cooldownRemaining = math.max(
					0,
					ACTION_COOLDOWN - (now - lastActionAt)
				)
			end

			local cooldownReady = cooldownRemaining <= 0
			local waveReady = expectedWave <= 0
				or currentWaveNumber >= expectedWave
			local moneyReady = currentGold >= requiredGold

			if cooldownReady
				and waveReady
				and moneyReady then
				return true
			end

			local waitingFor = {}

			if not cooldownReady then
				table.insert(
					waitingFor,
					string.format(
						"cooldown %.2f",
						cooldownRemaining
					)
				)
			end

			if not waveReady then
				table.insert(
					waitingFor,
					"wave "
						.. tostring(currentWaveNumber)
						.. "/"
						.. tostring(expectedWave)
				)
			end

			if not moneyReady then
				table.insert(
					waitingFor,
					"money "
						.. tostring(math.floor(currentGold))
						.. "/"
						.. tostring(math.floor(requiredGold))
				)
			end

			updateNextAction(
				action,
				actionIndex,
				cooldownRemaining,
				#waitingFor > 0
					and ("Waiting: " .. table.concat(waitingFor, ", "))
					or "Executing now"
			)

			task.wait(0.02)
		end

		return false,
			"The match ended before the action conditions were met."
	end

	local function playPlace(action)
		local unitId = tonumber(action.UnitId)

		if not unitId then
			return false, "Place action is missing UnitId."
		end

		if getRegisteredTower(unitId) then
			return false,
				"UnitId #" .. tostring(unitId) .. " is already bound."
		end

		if not waitForGold(action.Price) then
			return false, "The match ended while waiting for money."
		end

		local cframe = actionCFrame(action)
		local towerName = action.Template or action.Unit

		if not cframe or not towerName then
			return false, "Place action is missing CFrame or unit name."
		end

		local baseline = captureTowerSet()

		for attempt = 1, MAX_RETRIES do
			local delayedTower = waitForTowerResult(
				action,
				baseline,
				nil,
				1,
				nil,
				0.05
			)

			if delayedTower then
				return true
			end

			local success, result = pcall(function()
				return SpawnTower:InvokeServer(
					towerName,
					cframe
				)
			end)

			if success then
				local placed = waitForTowerResult(
					action,
					baseline,
					result,
					1,
					nil,
					SPAWN_CONFIRM_TIMEOUT
				)

				if placed then
					return true
				end
			end

			if not isGameActive() then
				break
			end

			task.wait(RETRY_DELAY)
		end

		return false,
			"Place did not create a new model for UnitId #"
				.. tostring(unitId)
	end

	local function playUpgrade(action)
		local targetLevel = tonumber(action.Level) or 2
		local baseline = captureTowerSet()

		for attempt = 1, MAX_RETRIES do
			local tower = findTower(
				action,
				{targetLevel, targetLevel - 1}
			)

			if not tower then
				task.wait(RETRY_DELAY)
			else
				if not towerMatchesAction(tower, action) then
					return false,
						"The bound Upgrade unit has a different name."
				end

				local config = tower:FindFirstChild("Config")
				local currentLevel = getTowerLevel(tower)

				if currentLevel >= targetLevel then
					bindTower(action.UnitId, tower)
					return true
				end

				if currentLevel ~= targetLevel - 1 then
					return false,
						"UnitId #"
							.. tostring(action.UnitId)
							.. " level does not match before Upgrade."
				end

				local upgrade =
					config and config:FindFirstChild("Upgrade")

				local nextTower = upgrade
					and upgrade:IsA("ObjectValue")
					and upgrade.Value
					or nil

				local root = getTowerRoot(tower)

				if nextTower and root then
					if not waitForGold(action.Price) then
						return false,
							"The match ended while waiting for money."
					end

					local success, result = pcall(function()
						return SpawnTower:InvokeServer(
							nextTower.Name,
							root.CFrame,
							tower
						)
					end)

					if success then
						local upgraded = waitForTowerResult(
							action,
							baseline,
							result,
							targetLevel,
							tower,
							SPAWN_CONFIRM_TIMEOUT
						)

						if upgraded then
							return true
						end
					end
				end

				task.wait(RETRY_DELAY)
			end

			if not isGameActive() then
				break
			end
		end

		return false,
			"Upgrade was not accepted or UnitId was lost."
	end

	local function playSell(action)
		for attempt = 1, MAX_RETRIES do
			local unitId = tonumber(action.UnitId)
			local wasBound = unitId and towerRegistry[unitId] ~= nil
			local tower = findTower(action, action.Level)

			if not tower then
				if wasBound then
					clearTowerBinding(unitId)
					return true
				end

				task.wait(RETRY_DELAY)
			else
				if not towerMatchesAction(tower, action) then
					return false,
						"The bound Sell unit has a different name."
				end

				local sellLevel = getTowerLevel(tower)

				if action.Level
					and sellLevel ~= tonumber(action.Level) then
					return false,
						"Unit level does not match before Sell."
				end

				local success, result = pcall(function()
					return SellTower:InvokeServer(tower)
				end)

				if success and result then
					local started = os.clock()

					while tower.Parent
						and os.clock() - started < 2 do
						task.wait(0.05)
					end

					if not tower.Parent then
						clearTowerBinding(unitId)
						return true
					end
				end

				task.wait(RETRY_DELAY)
			end
		end

		return false, "Sell was not accepted by the server."
	end

	local function playTarget(action)
		local desiredMode = tostring(action.Mode or "")

		if desiredMode == "" then
			return false, "Target action is missing a mode."
		end

		for attempt = 1, MAX_RETRIES do
			local tower = findTower(action, action.Level)

			if not tower then
				task.wait(RETRY_DELAY)
			else
				if not towerMatchesAction(tower, action) then
					return false,
						"The bound Target unit has a different name."
				end

				local config = tower:FindFirstChild("Config")
				local targetLevel =
					tonumber(readValue(config, "Level")) or 1

				if action.Level
					and targetLevel ~= tonumber(action.Level) then
					return false,
						"Unit level does not match before Target."
				end

				local currentMode = tostring(
					readValue(config, "TargetMode") or ""
				)

				if currentMode == desiredMode then
					return true
				end

				pcall(function()
					ChangeTowerMode:InvokeServer(tower)
				end)

				task.wait(0.15)
			end

			if not isGameActive() then
				break
			end
		end

		return false,
			"Could not set Target " .. desiredMode .. "."
	end

	local function getSkipButton()
		local gameGui =
			player.PlayerGui:FindFirstChild("GameGui")

		local skipFrame = gameGui
			and gameGui:FindFirstChild("SkipButton")

		local button = skipFrame
			and skipFrame:FindFirstChild("Skip")

		return button and button:IsA("GuiButton")
			and button
			or nil
	end

	local function playSkip(action)
		if not VoteSkip or not VoteSkip:IsA("RemoteEvent") then
			return false, "RemoteEvent VoteSkip was not found."
		end

		local expectedWave = tonumber(action.Wave) or 0

		if expectedWave > 0 and currentWave() > expectedWave then
			return true
		end

		local startingWave = currentWave()

		for attempt = 1, MAX_RETRIES do
			while isGameActive()
				and currentWave() == startingWave do
				local button = getSkipButton()

				if button
					and button.Visible
					and button.Interactable then
					break
				end

				task.wait(0.1)
			end

			if currentWave() ~= startingWave then
				return true
			end

			pcall(function()
				VoteSkip:FireServer()
			end)

			local confirmationStarted = os.clock()

			while isGameActive()
				and currentWave() == startingWave
				and os.clock() - confirmationStarted < 4 do
				local skipPlayers =
					tonumber(
						readValue(info, "SkipPlayers")
					) or 0

				local needed =
					tonumber(readValue(info, "Needed")) or 0

				if needed > 0 and skipPlayers >= needed then
					return true
				end

				task.wait(0.1)
			end

			if currentWave() ~= startingWave then
				return true
			end

			task.wait(RETRY_DELAY)
		end

		return false, "Skip was not confirmed."
	end

	local function playSpeed(action)
		if not Speed or not Speed:IsA("RemoteEvent") then
			return false, "RemoteEvent Speed was not found."
		end

		local speed = tonumber(action.Speed)

		if not speed then
			return false, "Speed action is missing a value."
		end

		local success = pcall(function()
			Speed:FireServer(speed)
		end)

		return success,
			success and nil or "Speed was not sent."
	end

	local function playAction(action)
		if action.Type == "Place" then
			return playPlace(action)
		elseif action.Type == "Upgrade" then
			return playUpgrade(action)
		elseif action.Type == "Sell" then
			return playSell(action)
		elseif action.Type == "Target" then
			return playTarget(action)
		elseif action.Type == "Skip" then
			return playSkip(action)
		elseif action.Type == "Speed" then
			return playSpeed(action)
		end

		return false,
			"Unknown action " .. tostring(action.Type)
	end

	local function handlePostGame()
		task.wait(0.75)

		if STRATEGY.AutoVipAfterGame then
			launchVipServer()
		elseif STRATEGY.AutoReplay then
			local exitGame = events:FindFirstChild("ExitGame")

			if exitGame and exitGame:IsA("RemoteEvent") then
				exitGame:FireServer()
				log("Replay vote sent.")
			end
		elseif STRATEGY.AutoBackToLobby then
			local lobbyTeleport =
				events:FindFirstChild("LobbyTeleport")

			if lobbyTeleport
				and lobbyTeleport:IsA("RemoteEvent") then
				lobbyTeleport:FireServer()
				log("Returning to the lobby.")
			end
		end
	end

	local function runStrategy()
		table.clear(towerRegistry)
		table.clear(towerUnitIds)

		local compatible, compatibilityError =
			checkCompatibility()

		if not compatible then
			fail(
				"Compatibility check failed: "
					.. tostring(compatibilityError),
				nil,
				nil,
				"Compatibility Check"
			)
			return false
		end

		if not waitForRecordingStartWave() then
			return false
		end

		local lastActionAt = nil

		log(
			"Starting: "
				.. tostring(#(STRATEGY.Actions or {}))
				.. " actions."
		)

		for index, action in ipairs(STRATEGY.Actions or {}) do
			if isActionGroupEnabled(action) then
				local timingOk, timingError =
					waitForAction(
						action,
						index,
						lastActionAt
					)

				if not timingOk then
					fail(
						"Action #"
							.. tostring(index)
							.. ": "
							.. tostring(timingError),
						action,
						index,
						"Wait For Action"
					)
					return false
				end

				updateNextAction(action, index, 0)

				local success, actionError = playAction(action)

				if not success then
					fail(
						"Action #"
							.. tostring(index)
							.. " ("
							.. tostring(action.Type)
							.. "): "
							.. tostring(actionError),
						action,
						index,
						"Play Action"
					)
					return false
				end

				lastActionAt = os.clock()

				log(
					"#"
						.. tostring(index)
						.. " "
						.. tostring(action.Type)
						.. " completed."
				)
			else
				log(
					"#"
						.. tostring(index)
						.. " skipped: group "
						.. tostring(action.Group)
						.. " is disabled."
				)
			end
		end

		updateNextAction(nil)
		log("Strategy completed successfully.")
		return true
	end

	while info.Parent do
		startMatchSelection()

		if not waitForActiveGame() then
			break
		end

		runStrategy()

		while info.Parent and isGameActive() do
			task.wait(0.5)
		end

		if info.Parent then
			handlePostGame()
		end

		task.wait(0.5)
	end

	environment.HamsterStrategyPlayerRunning = false
end

task.spawn(startHamsterStrategyPlayer)

local autosaveWriting = false
local autosavePending = false

saveAutosave = function()
	if not recording or typeof(writefile) ~= "function" then
		return false
	end

	if autosaveWriting then
		autosavePending = true
		return true
	end

	autosaveWriting = true

	repeat
		autosavePending = false

		local content = serializeStrategy()
		local success, writeError = executorWrite(AUTOSAVE_PATH, content)

		if not success then
			autosaveWriting = false
			setDebug(
			"Autosave",
				"Could not update the strategy autosave: "
					.. tostring(writeError),
				"error"
			)
			return false
		end

		task.wait()
	until not autosavePending or not recording

	autosaveWriting = false
	updateRayfieldLabel(
		"Autosave",
		"Autosave: " .. AUTOSAVE_PATH .. " • "
			.. tostring(#recordedActions)
			.. " actions"
	)
	return true
end

local function restoreAutosave()
	if typeof(readfile) ~= "function"
		or typeof(writefile) ~= "function"
		or typeof(isfile) ~= "function" then
		setDebug(
			"Autosave",
			"readfile, writefile, or isfile is unavailable.",
			"error"
		)
		return false
	end

	if not executorFileExists(AUTOSAVE_PATH) then
		setDebug(
			"Autosave",
			"Strat_Autosave.txt was not found.",
			"waiting"
		)
		return false
	end

	local success, errorMessage = pcall(function()
		writefile(
			STRATEGY_PATH,
			executorRead(AUTOSAVE_PATH) or ""
		)
	end)

	if not success then
		setDebug(
			"Autosave",
			"Could not restore autosave: "
				.. tostring(errorMessage),
			"error"
		)
		return false
	end

	setDebug(
		"Autosave",
		"Restored the strategy autosave to " .. STRATEGY_PATH .. ".",
		"success"
	)
	return true
end

local function saveStrategy()
	if typeof(writefile) ~= "function" then
		setDebug(
			"Save error",
			"writefile is unavailable. strat.txt was not saved.",
			"error"
		)
		return false
	end

	local content = serializeStrategy()
	local success, errorMessage = executorWrite(STRATEGY_PATH, content)

	if not success then
		setDebug(
			"Save error",
			"Could not save strat.txt: "
				.. tostring(errorMessage),
			"error"
		)
		return false
	end

	setDebug(
		"Saved",
		"Macro successfully saved to " .. STRATEGY_PATH,
		"success"
	)

	if typeof(delfile) == "function" then
		pcall(function()
			delfile(AUTOSAVE_PATH)
		end)
	end

	updateRayfieldLabel("Autosave", "Autosave cleared after saving")

	return true
end

local function startRecording()
	if recording then
		setDebug(
			"Recorder",
			"Recording is already running.",
			"waiting"
		)
		return
	end

	table.clear(recordedActions)
	table.clear(towerIds)
	nextTowerId = 1
	recordingStartedAt = os.clock()
	recording = true
	gameEndHandled = false
	recordingStartWave = getCurrentWave()
	recordingStartTiming = getCurrentTiming(0)
	recordingCreatedAt = os.date("!%Y-%m-%dT%H:%M:%SZ")
	lastRecordedSkipWave = nil
	lastRecordedSpeed = nil
	rememberCurrentTowers()
	updateRecorderStatus()

	if autoX2SpeedEnabled and requestX2Speed then
		if currentGameSpeed == 2 and appendSpeedAction then
			appendSpeedAction(2)
		else
			task.defer(requestX2Speed)
		end
	end

	setDebug(
		"Recorder",
		"Recording started. Capturing units, skips, and speed changes.",
		"success"
	)

	if saveAutosave then
		task.defer(saveAutosave)
	end
end

local function stopRecording(shouldSave, discardPendingSells)
	if not recording then
		setDebug(
			"Recorder",
			"Recording is already stopped.",
			"waiting"
		)
		return false
	end

	if discardPendingSells then
		table.clear(recentRemovedTowers)
	elseif flushPendingSells then
		task.wait(0.35)
		flushPendingSells()
	end

	recording = false
	updateRecorderStatus()

	if shouldSave then
		return saveStrategy()
	end

	setDebug(
		"Recorder",
		"Recording stopped.",
		"success"
	)

	return true
end

local function commitRecordedAction(action)
	if not recording or type(action) ~= "table" then
		return false
	end

	action.Group = nil
	table.insert(recordedActions, action)
	updateRecorderStatus()

	if saveAutosave then
		task.defer(saveAutosave)
	end

	return true
end

local function isUpgradeMatch(previousSnapshot, newSnapshot)
	if newSnapshot.Level > previousSnapshot.Level then
		return true
	end

	if previousSnapshot.NextPrice
		and newSnapshot.Price > 0 then
		return math.abs(
			newSnapshot.Price - previousSnapshot.NextPrice
		) < 0.01
	end

	return false
end

local function appendPlaceAction(snapshot)
	local elapsed = roundNumber(
		(snapshot.CapturedAt or os.clock()) - recordingStartedAt,
		3
	)
	local timing = snapshot.Timing ~= ""
		and snapshot.Timing
		or getCurrentTiming(elapsed)

	local action = {
		Type = "Place",
		UnitId = snapshot.UnitId,
		Time = elapsed,
		Wave = snapshot.Wave or getCurrentWave(),
		Unit = snapshot.Name,
		Template = snapshot.Template,
		Price = snapshot.Price,
		Timing = timing,
		Position = {
			X = roundNumber(snapshot.Position.X, 3),
			Y = roundNumber(snapshot.Position.Y, 3),
			Z = roundNumber(snapshot.Position.Z, 3),
		},
		CFrame = snapshot.CFrame,
	}

	commitRecordedAction(action)

	setDebug(
		"Recorder",
		"Placed "
			.. snapshot.Name
			.. " at "
			.. formatCoordinates(snapshot.Position),
		"success"
	)
end

local function appendUpgradeAction(snapshot, previousSnapshot)
	local price = previousSnapshot
		and previousSnapshot.NextPrice
		or snapshot.Price
		or 0

	local elapsed = roundNumber(
		(snapshot.CapturedAt or os.clock()) - recordingStartedAt,
		3
	)
	local timing = snapshot.Timing ~= ""
		and snapshot.Timing
		or getCurrentTiming(elapsed)

	local action = {
		Type = "Upgrade",
		UnitId = snapshot.UnitId
			or (previousSnapshot and previousSnapshot.UnitId),
		Time = elapsed,
		Wave = snapshot.Wave or getCurrentWave(),
		Unit = snapshot.Name,
		PreviousUnit = previousSnapshot
			and previousSnapshot.Name
			or nil,
		PreviousTemplate = previousSnapshot
			and previousSnapshot.Template
			or nil,
		Timing = timing,
		Position = {
			X = roundNumber(snapshot.Position.X, 3),
			Y = roundNumber(snapshot.Position.Y, 3),
			Z = roundNumber(snapshot.Position.Z, 3),
		},
		CFrame = snapshot.CFrame,
		Level = snapshot.Level,
		Price = price,
	}

	commitRecordedAction(action)

	setDebug(
		"Recorder",
		"Upgraded "
			.. snapshot.Name
			.. " to the next level for "
			.. tostring(price)
			.. "!",
		"success"
	)
end

local function appendSkipAction(waveNumber)
	if not recording
		or lastRecordedSkipWave == waveNumber then
		return
	end

	local elapsed = roundNumber(
		os.clock() - recordingStartedAt,
		3
	)
	local timing = getCurrentTiming(elapsed)

	lastRecordedSkipWave = waveNumber

	commitRecordedAction({
		Type = "Skip",
		Time = elapsed,
		Wave = waveNumber,
		Timing = timing,
	})

	setDebug(
		"Recorder",
		"Skipped wave "

		Wave = getCurrentWave(),

		Timing = timing,

			.. " at time "
			.. timing
			.. "!",
		"success"
	)

connectTowerTargetRecorder = function(tower)

		return
	end

		and config:FindFirstChild("TargetMode")

				)
			end
		end)
end

local function findUpgradeSource(newSnapshot)
	local deadline = os.clock() + 0.2

	repeat
		local bestIndex = nil
		local bestSnapshot = nil
		local bestDistance = TOWER_MATCH_DISTANCE

		for index = #recentRemovedTowers, 1, -1 do
			local removed = recentRemovedTowers[index]
			local age = os.clock() - removed.RemovedAt

			if age > UPGRADE_MATCH_WINDOW then
				table.remove(recentRemovedTowers, index)
			else
				local distance = (
					removed.Snapshot.Position
						- newSnapshot.Position
				).Magnitude

				if distance <= bestDistance
					and isUpgradeMatch(
						removed.Snapshot,
						newSnapshot
					) then
					bestIndex = index
					bestSnapshot = removed.Snapshot
					bestDistance = distance
				end
			end
		end

		if bestIndex then
			table.remove(recentRemovedTowers, bestIndex)
			return bestSnapshot
		end

		task.wait(0.03)
	until os.clock() >= deadline

	return nil
end

local function connectTowerRecorder()
	local function queueRemovedTower(tower, snapshot)
		if missingTowerStates[tower] then
			return
		end

		missingTowerStates[tower] = true

		local removedEntry = {
			Snapshot = snapshot,
			RemovedAt = os.clock(),
			Time = roundNumber(
				os.clock() - recordingStartedAt,
				3
			),
			Wave = getCurrentWave(),
		}

		removedEntry.Timing =
			getCurrentTiming(removedEntry.Time)

		table.insert(recentRemovedTowers, removedEntry)

		task.delay(UPGRADE_MATCH_WINDOW + 0.15, function()
			for index, entry in ipairs(recentRemovedTowers) do
				if entry == removedEntry then
					table.remove(recentRemovedTowers, index)

					local info =
						workspace:FindFirstChild("Info")

					if recording
						and getValue(
							info,
							"GameRunning"
						) == true then
						appendSellAction(
							snapshot,
							removedEntry
						)
					end

					break
				end
			end
		end)
	end

	local function cleanupTower(tower)
		local targetConnection =
			towerTargetConnections[tower]

		if targetConnection then
			targetConnection:Disconnect()
			towerTargetConnections[tower] = nil
		end

		missingTowerStates[tower] = nil
		consumedUpgradeSources[tower] = nil
	end

	while session.active do
		local towersFolder = workspace:FindFirstChild("Towers")

		if towersFolder then
			local currentSnapshots = {}

			for _, tower in ipairs(towersFolder:GetChildren()) do
				if tower:IsA("Model") then
					local snapshot =
						getTowerSnapshot(tower, true)

					if snapshot then
						currentSnapshots[tower] =
							snapshot
					end
				end
			end

			-- Process removals before additions.  During an upgrade Roblox
			-- replaces the old model with a new one in the same frame; the
			-- removed snapshot must be available to findUpgradeSource().
			for tower, previousSnapshot in pairs(
				knownTowerSnapshots
			) do
				if not currentSnapshots[tower]
					and tower.Parent ~= towersFolder then
					knownTowerSnapshots[tower] =
						nil
					cleanupTower(tower)

					local info =
						workspace:FindFirstChild("Info")

					if recording
						and previousSnapshot
						and getValue(
							info,
							"GameRunning"
						) == true then
						queueRemovedTower(
							tower,
							previousSnapshot
						)
					end
				end
			end

			for tower, snapshot in pairs(currentSnapshots) do
				missingTowerStates[tower] = nil
				local previousSnapshot =
					knownTowerSnapshots[tower]

				if not previousSnapshot then
					if recording then
						local upgradeSource =
							findUpgradeSource(snapshot)

						if upgradeSource
							and upgradeSource.UnitId then
							towerIds[tower] =
								upgradeSource.UnitId
							snapshot.UnitId =
								upgradeSource.UnitId
							appendUpgradeAction(
								snapshot,
								upgradeSource
							)
						else
							snapshot.UnitId =
								getOrAssignTowerId(tower)
							appendPlaceAction(snapshot)
						end
					else
						snapshot.UnitId =
							getOrAssignTowerId(tower)
					end
				elseif recording
					and snapshot.Level > previousSnapshot.Level then
					snapshot.UnitId =
						previousSnapshot.UnitId
					appendUpgradeAction(
						snapshot,
						previousSnapshot
					)
				end

				knownTowerSnapshots[tower] = snapshot
				connectTowerTargetRecorder(tower)
			end
		end

		task.wait(0.1)
	end
end


local function isGameActive()
	local info = workspace:FindFirstChild("Info")
	local gameRunning = getValue(info, "GameRunning")

local function getNativeSpeedFromGui()
	local fallbackSpeed = nil

	for _, guiName in ipairs({"Speed2x", "Speed3x"}) do
		local speedGui = playerGui:FindFirstChild(guiName)
		local frame = speedGui
			and speedGui:FindFirstChild("Frame")

		local status = frame
			and frame:FindFirstChild("Status")

		if status
			and (status:IsA("TextLabel")
				or status:IsA("TextButton")) then
			local speed = string.match(status.Text, "x(%d+)")

			if speed then
				local numericSpeed = tonumber(speed)

				if speedGui:IsA("ScreenGui")
					and speedGui.Enabled then
					return numericSpeed
				end

				fallbackSpeed = fallbackSpeed or numericSpeed
			end
		end
	end

	return fallbackSpeed
end

local function updateNativeSpeedGui(speed)
	for _, guiName in ipairs({"Speed2x", "Speed3x"}) do
		local speedGui = playerGui:FindFirstChild(guiName)
		local frame = speedGui
			and speedGui:FindFirstChild("Frame")

		local status = frame
			and frame:FindFirstChild("Status")

		if status
			and (status:IsA("TextLabel")
				or status:IsA("TextButton")) then
			status.Text = "x" .. tostring(speed) .. " Speed"
		end
	end
end

requestGameSpeed = function(speed, forceRequest)
	speed = tonumber(speed)

	if not speed or not isGameActive() then
		return false
	end

	if not forceRequest and currentGameSpeed == speed then
		return true
	end

	local now = os.clock()

	if not forceRequest
		and now - lastSpeedRequestAt < 1.5 then
		return false
	end

	local events = ReplicatedStorage:FindFirstChild("Events")
	local speedEvent = events and events:FindFirstChild("Speed")

	if not speedEvent or not speedEvent:IsA("RemoteEvent") then
		return false
	end

	lastSpeedRequestAt = now

	local success = pcall(function()
		speedEvent:FireServer(speed)
	end)

	return success
end

requestX2Speed = function()
	if not autoX2SpeedEnabled then
		return false
	end

	return requestGameSpeed(2, false)
end

local function getSkipButton()
	local gameGui = playerGui:FindFirstChild("GameGui")
	local skipFrame = gameGui
		and gameGui:FindFirstChild("SkipButton")

	local skipButton = skipFrame
		and skipFrame:FindFirstChild("Skip")

	if skipButton and skipButton:IsA("GuiButton") then
		return skipButton
	end

	return nil
end

local function tryAutoSkip()
	if not autoSkipEnabled or not isGameActive() then
		return
	end

	local waveNumber = getCurrentWave()

	if waveNumber <= 0
		or lastAutoSkipWave == waveNumber then
		return
	end

	local skipButton = getSkipButton()

	if not skipButton
		or not skipButton.Visible
		or not skipButton.Interactable then
		return
	end

	local events = ReplicatedStorage:FindFirstChild("Events")
	local voteSkip = events and events:FindFirstChild("VoteSkip")

	if not voteSkip or not voteSkip:IsA("RemoteEvent") then
		return
	end

	lastAutoSkipWave = waveNumber
	skipButton.Interactable = false
	skipButton.BackgroundColor3 =
		Color3.fromRGB(110, 110, 110)

	local success = pcall(function()
		voteSkip:FireServer()
	end)

	if not success then
		lastAutoSkipWave = nil
	end
end

local function checkConfirmedSkip(info)
	if not recording then
		return
	end

	local skipPlayers =
		tonumber(getValue(info, "SkipPlayers")) or 0

	local needed =
		tonumber(getValue(info, "Needed")) or 0

	local waveNumber =
		tonumber(getValue(info, "Wave")) or 0

	if needed > 0
		and skipPlayers >= needed
		and waveNumber > 0 then
		appendSkipAction(waveNumber)
	end
end

local function connectMatchAutomation()
	local info = workspace:WaitForChild("Info", 60)
	local events = ReplicatedStorage:WaitForChild("Events", 60)

	if not info or not events then
		return
	end

	currentGameSpeed = getNativeSpeedFromGui()
		or currentGameSpeed

	local speedEvent = events:FindFirstChild("Speed")

	if speedEvent and speedEvent:IsA("RemoteEvent") then
		speedEvent.OnClientEvent:Connect(function(_, speed)
			local confirmedSpeed = tonumber(speed)

			if not confirmedSpeed then
				return
			end

			currentGameSpeed = confirmedSpeed
			updateNativeSpeedGui(confirmedSpeed)
			appendSpeedAction(confirmedSpeed)

			if autoX2SpeedEnabled
				and confirmedSpeed ~= 2 then
				task.delay(0.35, requestX2Speed)
			end
		end)
	end

	local skipPlayers = info:FindFirstChild("SkipPlayers")
	local wave = info:FindFirstChild("Wave")

	if skipPlayers and skipPlayers:IsA("ValueBase") then
		local previousSkipCount =
			tonumber(skipPlayers.Value) or 0

		skipPlayers.Changed:Connect(function()
			local newSkipCount =
				tonumber(skipPlayers.Value) or 0

			if newSkipCount > previousSkipCount then
				checkConfirmedSkip(info)
			end

			previousSkipCount = newSkipCount
		end)
	end

	if wave and wave:IsA("ValueBase") then
		wave.Changed:Connect(function()
			if autoX2SpeedEnabled then
				task.defer(requestX2Speed)
			end
		end)
	end

	while session.active do
		tryAutoSkip()

		if autoX2SpeedEnabled then
			requestX2Speed()
		end

		task.wait(0.25)
	end
end


local function handleGameEnd()
	if gameEndHandled then
		return
	end

	gameEndHandled = true

	if recording then
		stopRecording(true, true)
	end

	task.delay(0.75, function()
		if autoVipAfterGameEnabled then
			launchVipServer()
			return
		end

		local events = ReplicatedStorage:FindFirstChild("Events")

		if not events then
			return
		end

		if autoReplayEnabled then
			local exitGame = events:FindFirstChild("ExitGame")

			if exitGame and exitGame:IsA("RemoteEvent") then
				exitGame:FireServer()
			end
		elseif autoBackToLobbyEnabled then
			local lobbyTeleport =
				events:FindFirstChild("LobbyTeleport")

			if lobbyTeleport
				and lobbyTeleport:IsA("RemoteEvent") then
				lobbyTeleport:FireServer()
			end
		end
	end)
end

			AutoTeleport = autoTeleportEnabled,

			AutoSkip = autoSkipEnabled,
			AutoX2Speed = autoX2SpeedEnabled,
			UpdatedAt = os.date("!%Y-%m-%dT%H:%M:%SZ"),
		})
	end)

	if success then
		local writeSuccess = executorWrite(
			SETTINGS_PATH,
			encoded
		)
		return writeSuccess == true
	end

	return false
end

local function validateExecutor()
	local missing = {}
	if typeof(loadstring) ~= "function" then table.insert(missing, "loadstring") end
	if typeof(game.HttpGet) ~= "function" then table.insert(missing, "game:HttpGet") end
	if typeof(readfile) ~= "function" then table.insert(missing, "readfile") end
	if typeof(writefile) ~= "function" then table.insert(missing, "writefile") end
	if typeof(isfile) ~= "function" then table.insert(missing, "isfile") end
	if typeof(makefolder) ~= "function" then table.insert(missing, "makefolder") end
	if #missing > 0 then
		local message = "Missing executor function(s): " .. table.concat(missing, ", ")
		warn("[Hamster Hub] " .. message)
		setDebug("Startup error", message, "error")
		return false
	end
	return true
end

local function scanLiveElevators()
	local lobby = workspace:FindFirstChild("LOBBY_")
	local elevators = lobby and lobby:FindFirstChild("Elevators")
	if not elevators then return "Searching for a solo elevator..." end
	local maps = {}
	local seenMaps = {}
	for _, elevator in ipairs(elevators:GetChildren()) do
		local mapLabel = elevator:FindFirstChild("mapName", true) or elevator:FindFirstChild("MapName", true)
		local map = mapLabel and ((mapLabel:IsA("TextLabel") or mapLabel:IsA("TextButton")) and mapLabel.Text or (mapLabel:IsA("StringValue") and mapLabel.Value))
		if typeof(map) == "string" and map ~= "" and not seenMaps[map] then
			seenMaps[map] = true
			table.insert(maps, map)
		end
	end
	if #maps == 0 then return "Searching for a solo elevator..." end
	table.sort(maps)
	return "Elevator scanner: " .. tostring(#maps) .. " map(s) detected"
end

local function readAutoexecStrategy()
	local registeredStrategy =
		environment.HamsterHubAutoexecStrategy

	if typeof(registeredStrategy) == "table" then
		return nil, "autoexecute:getgenv", registeredStrategy
	end

	for _, candidatePath in ipairs({
		"strat.txt",
		"Strat.txt",
		"autoexecute/strat.txt",
		"autoexecute/Strat.txt",
		"autoexec/strat.txt",
		"autoexec/Strat.txt",
		STRATEGY_PATH,
	}) do
		local content = executorRead(candidatePath)

		if content and content ~= "" then
			return content, candidatePath, nil
		end
	end

	return nil, nil, nil
end

local function startAutoExecute()
	if autoExecuteStarted then
		setDebug(
			"Auto Execute",
			"Strategy Player is already running.",
			"waiting"
		)
		return true
	end

	local content, strategyPath, registeredStrategy =
		readAutoexecStrategy()

	if not registeredStrategy
		and (not content or content == "") then
		setDebug(
			"Strategy Validator",
			"strat.txt did not register a strategy through autoexecute.",
			"error"
		)
		return false
	end

	local runtimeSource

	if registeredStrategy then
		runtimeSource =
			"local environment = typeof(getgenv) == \"function\" and getgenv() or _G\n"
			.. "local STRATEGY = environment.HamsterHubAutoexecStrategy\n"
			.. STRATEGY_PLAYER_SOURCE
	else
		local strategyDeclaration = content
		local oldPlayerStart = string.find(
			content,
			"\nlocal function startHamsterStrategyPlayer%("
		)

		if oldPlayerStart then
			strategyDeclaration = string.sub(
				content,
				1,
				oldPlayerStart - 1
			)
		end

		-- A recorder-generated file registers its table for the executor
		-- and returns it.  Those tail statements must not terminate the
		-- combined runtime before the current player source is appended.
		strategyDeclaration = strategyDeclaration
			:gsub("\nlocal environment = typeof%(%s*getgenv%s*%)%s*==%s*\"function\"%s*and%s*getgenv%(%s*%)%s*or%s*_G%s*", "")
			:gsub("\nenvironment%.HamsterHubAutoexecStrategy%s*=%s*STRATEGY%s*", "")
			:gsub("\nreturn%s+STRATEGY%s*$", "")

		runtimeSource =
			strategyDeclaration
			.. "\n"
			.. STRATEGY_PLAYER_SOURCE
	end

	local chunk, compileError =
		loadstring(runtimeSource)
	if not chunk then
		setDebug("Strategy Validator", "Invalid strategy format: " .. tostring(compileError), "error")
		return false
	end
	autoExecuteStarted = true
	setDebug("Auto Execute", "Searching for a solo elevator...", "waiting")
	local ok, runtimeError = pcall(chunk)
	if not ok then
		autoExecuteStarted = false
		appendFailureReport(runtimeError, {Stage = "Auto Execute"})
		setDebug("Auto Execute", tostring(runtimeError), "error")
		return false
	end
	return true
end

	destroyPreviousHubUi()
	local ok, result = pcall(function()

		local window = Rayfield:CreateWindow({

			LoadingTitle = HUB_TITLE,
			LoadingSubtitle = "Strategy automation",
			Theme = "Amethyst",

		mainTab:CreateSection("Automation")
		rayfieldToggles.AutoTeleport = mainTab:CreateToggle({Name = "Auto Teleport", CurrentValue = autoTeleportEnabled, Flag = "AutoTeleport", Callback = setAutoTeleport})
		rayfieldToggles.AutoReplay = mainTab:CreateToggle({Name = "Auto Replay", CurrentValue = autoReplayEnabled, Flag = "AutoReplay", Callback = setAutoReplay})

		rayfieldLabels.Loadout = mainTab:CreateLabel("Loadout: Unavailable")

		rayfieldLabels.StrategyValidator = strategyTab:CreateLabel("Strategy Validator: ready")
		rayfieldLabels.ElevatorScanner = strategyTab:CreateLabel("Searching for a solo elevator...")
		strategyTab:CreateParagraph({Title = "Supported Modes", Content = "Easy | Medium | Hard | Nightmare | Crazy | Boss Rush"})

		recorderTab:CreateSection("Macro Recorder")
		recorderTab:CreateButton({Name = "Start Recording", Callback = startRecording})
		recorderTab:CreateButton({Name = "Stop & Save", Callback = function() stopRecording(true) end})
		recorderTab:CreateButton({Name = "Restore Autosave", Callback = restoreAutosave})
		rayfieldLabels.RecorderStatus = recorderTab:CreateLabel("Stopped | Actions: 0")
		rayfieldLabels.Autosave = recorderTab:CreateLabel("Autosave: " .. AUTOSAVE_PATH)

		debugTab:CreateSection("Status")
		rayfieldLabels.Debug = debugTab:CreateLabel(lastDebugTitle .. ": " .. lastDebugMessage)
		rayfieldLabels.Failure = debugTab:CreateLabel("Failure log: " .. ERROR_LOG_PATH)
		debugTab:CreateLabel("Settings: " .. SETTINGS_PATH)

		configsTab:CreateSection("Settings")
		rayfieldToggles.ConfigAutoload = configsTab:CreateToggle({
			Name = "Auto Load Config",
			CurrentValue = configAutoloadEnabled,
			Flag = "ConfigAutoload",
			Callback = setConfigAutoload,
		})
		configsTab:CreateButton({
			Name = "Save Config",
			Callback = saveConfigFile,
		})
		configsTab:CreateButton({
			Name = "Load Config",
			Callback = loadConfigFile,
		})
		rayfieldLabels.ConfigStatus = configsTab:CreateLabel(
			"Config path: " .. SETTINGS_PATH
		)

		rayfieldLibrary = Rayfield
		renderMiscSettings()
		updateRecorderStatus()
		renderProfile()
		task.spawn(function()
			while session.active and rayfieldLibrary == Rayfield do
				updateRayfieldLabel("ElevatorScanner", scanLiveElevators())
				task.wait(1)
			end
		end)
		setDebug("Ready", "Rayfield interface loaded successfully.", "success")
	end)
	if not ok then
		local message = "Rayfield failed to load. Check executor HTTP support. " .. tostring(result)
		warn("[Hamster Hub] " .. message)
		setDebug("Rayfield", message, "error")
		return false
	end
	return true
end

task.spawn(function()
	local startedAt = os.clock()
	local autoexecStrategyPath

	repeat
		task.wait(0.25)
		local _, detectedPath =
			readAutoexecStrategy()
		autoexecStrategyPath = detectedPath
	until autoexecStrategyPath
		or not session.active
		or os.clock() - startedAt >= 10

	if session.active and autoexecStrategyPath then
		startAutoExecute()
	end
end)

task.spawn(function()
	local info = workspace:WaitForChild("Info", 60)
	if not info then return end
	local gameRunning = info:FindFirstChild("GameRunning")
	if gameRunning and gameRunning:IsA("BoolValue") then
		track(gameRunning:GetPropertyChangedSignal("Value"):Connect(function()
			if gameRunning.Value then
				gameEndHandled = false
				lastAutoSkipWave = nil
				currentGameSpeed = 1
				lastSpeedRequestAt = 0
				if autoX2SpeedEnabled then task.defer(requestX2Speed) end
			else
				handleGameEnd()
			end
		end))
	end
end)

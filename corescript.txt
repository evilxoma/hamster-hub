local function loadSavedSnapshot()
	local success, savedValue = pcall(function()
		return TeleportService:GetTeleportSetting(SNAPSHOT_KEY)
	end)

	if not success or typeof(savedValue) ~= "string" or savedValue == "" then
		return
	end

	local decodedSuccess, decoded = pcall(function()
		return HttpService:JSONDecode(savedValue)
	end)

	if not decodedSuccess or typeof(decoded) ~= "table" then
		return
	end

	if decoded.Coins ~= nil then
		profileSnapshot.Coins = tostring(decoded.Coins)
	end

	if decoded.Gems ~= nil then
		profileSnapshot.Gems = tostring(decoded.Gems)
	end

	if typeof(decoded.Settings) == "table" then
		profileSnapshot.Settings.AutoReplay =
			decoded.Settings.AutoReplay == true

		profileSnapshot.Settings.AutoBackToLobby =
			decoded.Settings.AutoBackToLobby == true

		profileSnapshot.Settings.AutoSkip =
			decoded.Settings.AutoSkip == true

		profileSnapshot.Settings.AutoX2Speed =
			decoded.Settings.AutoX2Speed == true

		if table.find(
			ACTION_GROUP_ORDER,
			tostring(decoded.Settings.CurrentActionGroup)
		) then
			profileSnapshot.Settings.CurrentActionGroup =
				tostring(decoded.Settings.CurrentActionGroup)
		end

		if typeof(decoded.Settings.EnabledGroups) == "table" then
			for _, groupName in ipairs(ACTION_GROUP_ORDER) do
				local storedValue =
					decoded.Settings.EnabledGroups[groupName]

				if storedValue ~= nil then
					profileSnapshot.Settings.EnabledGroups[groupName] =
						storedValue == true
				end
			end
		end

		if profileSnapshot.Settings.AutoReplay then
			profileSnapshot.Settings.AutoBackToLobby = false
		end
	end

	if typeof(decoded.Deck) == "table" then
		for slotNumber = 1, 6 do
			local savedSlot = decoded.Deck[slotNumber]
				or decoded.Deck[tostring(slotNumber)]

			if typeof(savedSlot) == "table" then
				profileSnapshot.Deck[slotNumber] = {
					Slot = slotNumber,
					Image = tostring(savedSlot.Image or ""),
					Name = tostring(savedSlot.Name or ""),
				}
			end
		end
	end

	savedSnapshotLoaded = true
end

loadSavedSnapshot()

topBar.InputBegan:Connect(function(input)
	if input.UserInputType ~= Enum.UserInputType.MouseButton1
		and input.UserInputType ~= Enum.UserInputType.Touch then
		return
	end

	dragging = true
	dragStart = input.Position
	startPosition = container.Position

	input.Changed:Connect(function()
		if input.UserInputState == Enum.UserInputState.End then
			dragging = false
		end
	end)
end)

topBar.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch then
		dragInput = input
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if not dragging or input ~= dragInput then
		return
	end

	if not dragStart or not startPosition then
		return
	end

	local delta = input.Position - dragStart

	container.Position = UDim2.new(
		startPosition.X.Scale,
		startPosition.X.Offset + delta.X,
		startPosition.Y.Scale,
		startPosition.Y.Offset + delta.Y
	)
end)


local towerNamesByImage = {}

local function normalizeImage(image)
	local text = tostring(image or "")
	local id = string.match(text, "%d+")
	return id or text
end

pcall(function()
	local moduleObject = ReplicatedStorage:FindFirstChild("Module_Towers")

	if moduleObject and moduleObject:IsA("ModuleScript") then
		local moduleData = require(moduleObject)

		if typeof(moduleData) == "table" then
			for towerName, towerInfo in pairs(moduleData) do
				if typeof(towerInfo) == "table" and towerInfo.Image then
					towerNamesByImage[normalizeImage(towerInfo.Image)] =
						tostring(towerName)
				end
			end
		end
	end
end)


local function findDeckSlotsContainer()
	local mainGui = playerGui:FindFirstChild("Main")

	if mainGui then
		local mainFrameObject = mainGui:FindFirstChild("Main")

		if mainFrameObject then
			local slotsOuter = mainFrameObject:FindFirstChild("Slots")
			local slotsInner = slotsOuter
				and slotsOuter:FindFirstChild("Slots")

			if slotsInner and slotsInner:FindFirstChild("Slot1") then
				return slotsInner
			end
		end
	end

	local gameGui = playerGui:FindFirstChild("GameGui")

	if gameGui then
		local slotsOuter = gameGui:FindFirstChild("Slots")
		local slotsInner = slotsOuter
			and slotsOuter:FindFirstChild("Slots")

		if slotsInner and slotsInner:FindFirstChild("Slot1") then
			return slotsInner
		end
	end

	for _, object in ipairs(playerGui:GetDescendants()) do
		if object.Name == "Slots"
			and object:FindFirstChild("Slot1")
			and object:FindFirstChild("Slot2") then
			return object
		end
	end

	return nil
end

local function getSlotImage(slotObject)
	if slotObject:IsA("ImageButton")
		or slotObject:IsA("ImageLabel") then
		if slotObject.Image and slotObject.Image ~= "" then
			return slotObject.Image
		end
	end

	local iconImage = slotObject:FindFirstChild("IconImage", true)

	if iconImage
		and (iconImage:IsA("ImageButton")
			or iconImage:IsA("ImageLabel"))
		and iconImage.Image ~= "" then
		return iconImage.Image
	end

	for _, descendant in ipairs(slotObject:GetDescendants()) do
		if descendant:IsA("ImageButton")
			or descendant:IsA("ImageLabel") then
			if descendant.Image ~= ""
				and not string.find(
					string.lower(descendant.Name),
					"shiny"
				) then
				return descendant.Image
			end
		end
	end

	return ""
end

local function getSlotUnitName(slotObject, image)
	local attributeNames = {
		"TowerName",
		"UnitName",
		"Name",
	}

	for _, attributeName in ipairs(attributeNames) do
		local value = slotObject:GetAttribute(attributeName)

		if typeof(value) == "string"
			and value ~= ""
			and value ~= slotObject.Name then
			return value
		end
	end

	local mappedName = towerNamesByImage[normalizeImage(image)]

	if mappedName then
		return mappedName
	end

	for _, descendant in ipairs(slotObject:GetDescendants()) do

			local lowerName = string.lower(descendant.Name)

			if string.find(lowerName, "tower")
				or string.find(lowerName, "unit") then
				if descendant.Value ~= "" then
					return descendant.Value
				end
			end
		end
	end

	return ""
end

local function roundNumber(value, digits)
	local multiplier = 10 ^ (digits or 0)
	return math.round(value * multiplier) / multiplier
end

local function getValue(containerObject, childName)
	local valueObject = containerObject
		and containerObject:FindFirstChild(childName)

		return valueObject.Value
	end

	return nil

local function getTowerCFrame(tower)
	if not tower or not tower:IsA("Model") then
		return nil
	end

	local success, pivot = pcall(function()
		return tower:GetPivot()
	end)

	if success then
		return pivot
	end

	local root = tower.PrimaryPart
		or tower:FindFirstChild("HumanoidRootPart")
		or tower:FindFirstChildWhichIsA("BasePart", true)

	return root and root.CFrame or nil
end

local function isOurTower(config)
	local owner = getValue(config, "Owner")

	if typeof(owner) == "Instance" and owner:IsA("Player") then
		return owner == player
	end

	return tostring(owner or "") == player.Name
end

local function getOrAssignTowerId(tower, preferredId)
	if not tower then
		return nil
	end

	local existingId = towerIds[tower]

	if existingId then
		return existingId
	end

	local unitId = tonumber(preferredId)

	if not unitId then
		unitId = nextTowerId
		nextTowerId = nextTowerId + 1
	else
		nextTowerId = math.max(nextTowerId, unitId + 1)
	end

	towerIds[tower] = unitId
	return unitId
end

local function getTowerSnapshot(tower, deferUnitId)
	if not tower or not tower:IsA("Model") then
		return nil
	end

	local config = tower:FindFirstChild("Config")

	if not config then
		config = tower:WaitForChild("Config", 2)
	end

	if not config or not isOurTower(config) then
		return nil
	end

	local towerCFrame = getTowerCFrame(tower)

	if not towerCFrame then
		return nil
	end

	local towerName = getValue(config, "TowerName")
		or getValue(config, "UnitName")
		or tower.Name

	local level = tonumber(getValue(config, "Level")) or 1
	local currentPrice =
		tonumber(getValue(config, "Price")) or 0

	local nextPrice = nil
	local upgradeObject = config:FindFirstChild("Upgrade")

	if upgradeObject and upgradeObject:IsA("ObjectValue") then
		local nextTower = upgradeObject.Value
		local nextConfig =
			nextTower and nextTower:FindFirstChild("Config")

		nextPrice = tonumber(getValue(nextConfig, "Price"))
	end

	local components = {towerCFrame:GetComponents()}

	for index, component in ipairs(components) do
		components[index] = roundNumber(component, 6)
	end

	local captureInfo = workspace:FindFirstChild("Info")
	local capturedTiming = getValue(captureInfo, "Time")

	return {
		Model = tower,
		UnitId = deferUnitId
			and towerIds[tower]
			or (
				not deferUnitId
				and getOrAssignTowerId(tower)
				or nil
			),
		Name = tostring(towerName),
		Template = tower.Name,
		Level = level,
		Price = currentPrice,
		SellValue = math.floor(currentPrice / 2),
		NextPrice = nextPrice,
		TargetMode = tostring(
			getValue(config, "TargetMode") or "First"
		),
		Position = towerCFrame.Position,
		CFrame = components,
		Wave = tonumber(getValue(captureInfo, "Wave")) or 0,
		Timing = capturedTiming ~= nil
			and tostring(capturedTiming)
			or "",
		CapturedAt = os.clock(),
	}
end

local function formatCoordinates(position)
	return string.format(
		"(%.2f, %.2f, %.2f)",
		position.X,
		position.Y,

end

local function formatTiming(seconds)
	local safeSeconds = math.max(0, tonumber(seconds) or 0)
	local minutes = math.floor(safeSeconds / 60)
	local remainingSeconds = safeSeconds - minutes * 60

	return string.format(
		"%02d:%05.2f",
		minutes,
		remainingSeconds
	)
end

	local gameTime = getValue(info, "Time")

	if gameTime ~= nil and tostring(gameTime) ~= "" then
		return tostring(gameTime)
	end

	return formatTiming(fallbackSeconds)
end

			container and container:FindFirstChild(name)

		if valueObject and valueObject:IsA("ValueBase") then
			return valueObject.Value
		end

		return nil
	end

	local function isGameActive()
		return readValue(info, "GameRunning") == true
	end

	local function currentWave()
		return tonumber(readValue(info, "Wave")) or 0
	end

	local function startModeVoting()
		local voteForMap = events:FindFirstChild("VoteForMap")

		if not voteForMap
			or not voteForMap:IsA("RemoteEvent")
			or not STRATEGY.Mode
			or STRATEGY.Mode == "Unknown" then
			return
		end

		task.spawn(function()
			local lastVoteAt = 0

			while info.Parent and not isGameActive() do
				if readValue(info, "Voting") == true
					and os.clock() - lastVoteAt >= 1 then
					lastVoteAt = os.clock()

					pcall(function()
						voteForMap:FireServer(STRATEGY.Mode)
					end)
				end

				task.wait(0.2)
			end
		end)
	end

	local function waitForActiveGame()
		while info.Parent and not isGameActive() do
			task.wait(0.25)
		end

		return info.Parent ~= nil
	end

	local function waitForRecordingStartWave()
		local expectedWave =
			tonumber(STRATEGY.StartWave) or 0

		while isGameActive()
			and currentWave() < expectedWave do
			task.wait(0.1)
		end

		if not isGameActive() then
			return false
		end

		if expectedWave > 0
			and currentWave() > expectedWave then
			fail(
				"Матч уже дошёл до волны "
					.. tostring(currentWave())
					.. ", а стратегия начинается с "
					.. tostring(expectedWave)
					.. "."
			)
			return false
		end

		return true
	end

	local function checkCompatibility()
		local metadata = typeof(STRATEGY.Metadata) == "table"
			and STRATEGY.Metadata
			or {}

		local formatVersion = tonumber(
			metadata.FormatVersion
				or STRATEGY.FormatVersion
				or 1
		) or 1

		if formatVersion > SUPPORTED_FORMAT_VERSION then
			return false,
				"Формат стратегии "
					.. tostring(formatVersion)
					.. " новее поддерживаемого "
					.. tostring(SUPPORTED_FORMAT_VERSION)
					.. "."
		end

		local mapName = tostring(
			readValue(info, "Map") or ""
		)

		local modeName = tostring(
			readValue(info, "Mode") or ""
		)

		local difficultyName = tostring(
			readValue(info, "Difficulty")
				or readValue(info, "DifficultyName")
				or readValue(info, "DifficultyMode")
				or ""
		)

		if STRATEGY.Map
			and STRATEGY.Map ~= "Unknown"
			and mapName ~= STRATEGY.Map then
			return false,
				"Карта не совпадает. Нужна "
					.. STRATEGY.Map
					.. ", сейчас "
					.. mapName
		end

		if STRATEGY.Mode
			and STRATEGY.Mode ~= "Unknown"
			and modeName ~= STRATEGY.Mode then
			return false,
				"Режим не совпадает. Нужен "
					.. STRATEGY.Mode
					.. ", сейчас "
					.. modeName
		end

		if STRATEGY.Difficulty
			and STRATEGY.Difficulty ~= "Unknown"
			and STRATEGY.Difficulty ~= "Any"
			and difficultyName ~= ""
			and difficultyName ~= STRATEGY.Difficulty then
			return false,
				"Сложность не совпадает. Нужна "
					.. STRATEGY.Difficulty
					.. ", сейчас "
					.. difficultyName
		end

		local success, data = pcall(function()
			return GetData:InvokeServer()
		end)

		if not success or typeof(data) ~= "table" then
			return false,
				"Сервер не вернул данные текущей колоды."
		end

		local deckNames = {}

		for slotNumber = 1, 6 do
			local slot =
				data["Slot" .. tostring(slotNumber)]

			local first = typeof(slot) == "table"
				and slot[1]
				or nil

			local unitName = typeof(first) == "table"
				and (first.name or first.Name)
				or nil

			if unitName then
				deckNames[
					string.lower(tostring(unitName))
				] = true
			end
		end

		for _, action in ipairs(STRATEGY.Actions or {}) do
			if isActionGroupEnabled(action)
				and action.Type == "Place" then
				local requiredName = tostring(
					action.Template or action.Unit or ""
				)

				if requiredName ~= ""
					and not deckNames[
						string.lower(requiredName)
					] then
					return false,
						"В колоде нет юнита "
							.. requiredName
							.. "."
				end
			end
		end

		if formatVersion >= 4 then
			local placedUnitIds = {}

			for index, action in ipairs(STRATEGY.Actions or {}) do
				local actionType = tostring(action.Type or "")
				local needsUnitId = actionType == "Place"
					or actionType == "Upgrade"
					or actionType == "Sell"
					or actionType == "Target"
				local unitId = tonumber(action.UnitId)

				if needsUnitId and not unitId then
					return false,
						"В действии #"
							.. tostring(index)
							.. " отсутствует UnitId."
				end

				if actionType == "Place" then
					if placedUnitIds[unitId] then
						return false,
							"UnitId #"
								.. tostring(unitId)
								.. " используется несколькими Place."
					end

					placedUnitIds[unitId] = true
				elseif needsUnitId and not placedUnitIds[unitId] then
					return false,
						"Действие #"
							.. tostring(index)
							.. " ссылается на UnitId до Place."
				end
			end
		end

		return true
	end

	local function getTowerRoot(tower)
		if not tower or not tower:IsA("Model") then
			return nil
		end

		return tower.PrimaryPart
			or tower:FindFirstChild("HumanoidRootPart")
			or tower:FindFirstChildWhichIsA(
				"BasePart",
				true
			)
	end

	local function isOurTower(tower)
		local config =
			tower and tower:FindFirstChild("Config")

		local owner = readValue(config, "Owner")

		if typeof(owner) == "Instance"
			and owner:IsA("Player") then
			return owner == player
		end

		return tostring(owner or "") == player.Name
	end

	local function actionPosition(action)
		if not action.Position then
			return nil
		end

		return Vector3.new(
			tonumber(action.Position.X) or 0,
			tonumber(action.Position.Y) or 0,
			tonumber(action.Position.Z) or 0
		)
	end

	local function actionCFrame(action)
		if typeof(action.CFrame) == "table"
			and #action.CFrame >= 12 then
			return CFrame.new(table.unpack(action.CFrame))
		end

		local position = actionPosition(action)
		return position and CFrame.new(position) or nil
	end

	local function normalizedName(value)
		return string.lower(tostring(value or ""))
	end

	local function towerMatchesAction(tower, action)
		if not tower then
			return false
		end

		local config = tower:FindFirstChild("Config")
		local towerName = normalizedName(
			readValue(config, "TowerName")
				or readValue(config, "UnitName")
				or tower.Name
		)
		local templateName = normalizedName(tower.Name)
		local expectedNames = {
			action.Unit,
			action.PreviousUnit,
			action.Template,
			action.PreviousTemplate,
		}
		local hasExpectedName = false

		for _, expected in ipairs(expectedNames) do
			local normalizedExpected = normalizedName(expected)

			if normalizedExpected ~= "" then
				hasExpectedName = true

				if towerName == normalizedExpected
					or templateName == normalizedExpected then
					return true
				end
			end
		end

		return not hasExpectedName
	end

	local function getTowerLevel(tower)
		local config = tower and tower:FindFirstChild("Config")
		return tonumber(readValue(config, "Level")) or 1
	end

	local function clearTowerBinding(unitId)
		unitId = tonumber(unitId)

		if not unitId then
			return
		end

		local tower = towerRegistry[unitId]
		towerRegistry[unitId] = nil

		if tower and towerUnitIds[tower] == unitId then
			towerUnitIds[tower] = nil
		end
	end

	local function bindTower(unitId, tower)
		unitId = tonumber(unitId)

		if not unitId or not tower or not tower:IsA("Model") then
			return tower
		end

		local previousTower = towerRegistry[unitId]

		if previousTower and previousTower ~= tower
			and towerUnitIds[previousTower] == unitId then
			towerUnitIds[previousTower] = nil
		end

		local previousUnitId = towerUnitIds[tower]

		if previousUnitId and previousUnitId ~= unitId then
			towerRegistry[previousUnitId] = nil
		end

		towerRegistry[unitId] = tower
		towerUnitIds[tower] = unitId
		return tower
	end

	local function getRegisteredTower(unitId)
		unitId = tonumber(unitId)

		if not unitId then
			return nil
		end

		local tower = towerRegistry[unitId]

		if tower
			and tower.Parent
			and tower:IsA("Model")
			and isOurTower(tower) then
			return tower
		end

		clearTowerBinding(unitId)
		return nil
	end

	local function captureTowerSet()
		local result = {}
		local towers = workspace:FindFirstChild("Towers")

		if towers then
			for _, tower in ipairs(towers:GetChildren()) do
				if tower:IsA("Model") then
					result[tower] = true
				end
			end
		end

		return result
	end

	local function levelIsAllowed(tower, allowedLevels)
		if allowedLevels == nil then
			return true
		end

		local currentLevel = getTowerLevel(tower)

		if typeof(allowedLevels) == "number" then
			return currentLevel == allowedLevels
		end

		if typeof(allowedLevels) == "table" then
			for _, level in ipairs(allowedLevels) do
				if currentLevel == tonumber(level) then
					return true
				end
			end
		end

		return false
	end

	local function findFallbackTower(action, allowedLevels)
		local towers = workspace:FindFirstChild("Towers")
		local position = actionPosition(action)
		local unitId = tonumber(action.UnitId)

		if not towers or not position then
			return nil
		end

		local closest = nil
		local closestDistance = POSITION_TOLERANCE

		for _, tower in ipairs(towers:GetChildren()) do
			if tower:IsA("Model")
				and isOurTower(tower)
				and towerMatchesAction(tower, action)
				and levelIsAllowed(tower, allowedLevels) then
				local boundId = towerUnitIds[tower]

				if not boundId or boundId == unitId then
					local root = getTowerRoot(tower)

					if root then
						local distance =
							(root.Position - position).Magnitude

						if distance <= closestDistance then
							closest = tower
							closestDistance = distance
						end
					end
				end
			end
		end

		if closest and unitId then
			bindTower(unitId, closest)
		end

		return closest
	end

	local function findTower(action, allowedLevels)
		local registered = getRegisteredTower(action.UnitId)

		if registered then
			return registered
		end

		return findFallbackTower(action, allowedLevels)
	end

	local function waitForTowerResult(
		action,
		baseline,
		returnedValue,
		expectedLevel,
		allowExistingTower,
		timeout
	)
		local towers = workspace:FindFirstChild("Towers")

		if not towers then
			return nil
		end

		local returnedTower = typeof(returnedValue) == "Instance"
			and returnedValue:IsA("Model")
			and returnedValue
			or nil
		local expectedPosition = actionPosition(action)
		local unitId = tonumber(action.UnitId)
		local deadline = os.clock() + (timeout or SPAWN_CONFIRM_TIMEOUT)

		repeat
			local candidates = {}

			if returnedTower and returnedTower.Parent then
				table.insert(candidates, returnedTower)
			end

			for _, tower in ipairs(towers:GetChildren()) do
				if tower:IsA("Model") and tower ~= returnedTower then
					table.insert(candidates, tower)
				end
			end

			for _, tower in ipairs(candidates) do
				local isNewTower = not baseline[tower]
				local isAllowedExisting =
					allowExistingTower ~= nil
					and tower == allowExistingTower

				if (isNewTower or isAllowedExisting)
					and tower.Parent
					and isOurTower(tower)
					and towerMatchesAction(tower, action)
					and getTowerLevel(tower) == expectedLevel then
					local boundId = towerUnitIds[tower]
					local root = getTowerRoot(tower)
					local closeEnough = root ~= nil

					if root and expectedPosition then
						closeEnough =
							(root.Position - expectedPosition).Magnitude
							<= POSITION_TOLERANCE
					end

					if closeEnough
						and (not boundId or boundId == unitId) then
						return bindTower(unitId, tower)
					end
				end
			end

			task.wait(0.05)
		until os.clock() >= deadline or not isGameActive()

		return nil
	end

	local function waitForGold(requiredAmount)
		local required = tonumber(requiredAmount) or 0

		if required <= 0 then
			return true
		end

		local gold = player:FindFirstChild("Gold")

		while isGameActive() do
			if gold
				and tonumber(gold.Value)
				and tonumber(gold.Value) >= required then
				return true
			end

			task.wait(0.1)
			gold = gold or player:FindFirstChild("Gold")
		end

		return false
	end

	local function waitForAction(action, startedAt, actionIndex)
		local targetTime = tonumber(action.Time) or 0

		while isGameActive()
			and os.clock() - startedAt < targetTime do
			updateNextAction(
				action,
				actionIndex,
				targetTime - (os.clock() - startedAt)
			)
			task.wait(0.05)
		end

		if not isGameActive() then
			return false, "Матч закончился раньше действия."
		end

		local expectedWave = tonumber(action.Wave) or 0

		while isGameActive()
			and expectedWave > 0
			and currentWave() < expectedWave do
			updateNextAction(action, actionIndex, 0)
			task.wait(0.05)
		end

		if not isGameActive() then
			return false, "Матч закончился раньше нужной волны."
		end

		return true
	end

	local function playPlace(action)
		local unitId = tonumber(action.UnitId)

		if not unitId then
			return false, "В Place отсутствует UnitId."
		end

		if getRegisteredTower(unitId) then
			return false,
				"UnitId #" .. tostring(unitId) .. " уже привязан."
		end

		if not waitForGold(action.Price) then
			return false, "Матч завершился во время ожидания денег."
		end

		local cframe = actionCFrame(action)
		local towerName = action.Template or action.Unit

		if not cframe or not towerName then
			return false, "В Place отсутствуют CFrame или имя."
		end

		local baseline = captureTowerSet()

		for attempt = 1, MAX_RETRIES do
			local delayedTower = waitForTowerResult(
				action,
				baseline,
				nil,
				1,
				nil,
				0.05
			)

			if delayedTower then
				return true
			end

			local success, result = pcall(function()
				return SpawnTower:InvokeServer(
					towerName,
					cframe
				)
			end)

			if success then
				local placed = waitForTowerResult(
					action,
					baseline,
					result,
					1,
					nil,
					SPAWN_CONFIRM_TIMEOUT
				)

				if placed then
					return true
				end
			end

			if not isGameActive() then
				break
			end

			task.wait(RETRY_DELAY)
		end

		return false,
			"Place не создал новую модель для UnitId #"
				.. tostring(unitId)
	end

	local function playUpgrade(action)
		local targetLevel = tonumber(action.Level) or 2
		local baseline = captureTowerSet()

		for attempt = 1, MAX_RETRIES do
			local tower = findTower(
				action,
				{targetLevel, targetLevel - 1}
			)

			if not tower then
				task.wait(RETRY_DELAY)
			else
				if not towerMatchesAction(tower, action) then
					return false,
						"Привязанный Upgrade-юнит имеет другое имя."
				end

				local config = tower:FindFirstChild("Config")
				local currentLevel = getTowerLevel(tower)

				if currentLevel >= targetLevel then
					bindTower(action.UnitId, tower)
					return true
				end

				if currentLevel ~= targetLevel - 1 then
					return false,
						"Уровень UnitId #"
							.. tostring(action.UnitId)
							.. " не совпадает перед Upgrade."
				end

				local upgrade =
					config and config:FindFirstChild("Upgrade")

				local nextTower = upgrade
					and upgrade:IsA("ObjectValue")
					and upgrade.Value
					or nil

				local root = getTowerRoot(tower)

				if nextTower and root then
					if not waitForGold(action.Price) then
						return false,
							"Матч завершился при ожидании денег."
					end

					local success, result = pcall(function()
						return SpawnTower:InvokeServer(
							nextTower.Name,
							root.CFrame,
							tower
						)
					end)

					if success then
						local upgraded = waitForTowerResult(
							action,
							baseline,
							result,
							targetLevel,
							tower,
							SPAWN_CONFIRM_TIMEOUT
						)

						if upgraded then
							return true
						end
					end
				end

				task.wait(RETRY_DELAY)
			end

			if not isGameActive() then
				break
			end
		end

		return false,
			"Upgrade не был принят или UnitId потерян."
	end

	local function playSell(action)
		for attempt = 1, MAX_RETRIES do
			local unitId = tonumber(action.UnitId)
			local wasBound = unitId and towerRegistry[unitId] ~= nil
			local tower = findTower(action, action.Level)

			if not tower then
				if wasBound then
					clearTowerBinding(unitId)
					return true
				end

				task.wait(RETRY_DELAY)
			else
				if not towerMatchesAction(tower, action) then
					return false,
						"Привязанный Sell-юнит имеет другое имя."
				end

				local sellLevel = getTowerLevel(tower)

				if action.Level
					and sellLevel ~= tonumber(action.Level) then
					return false,
						"Уровень юнита не совпадает перед Sell."
				end

				local success, result = pcall(function()
					return SellTower:InvokeServer(tower)
				end)

				if success and result then
					local started = os.clock()

					while tower.Parent
						and os.clock() - started < 2 do
						task.wait(0.05)
					end

					if not tower.Parent then
						clearTowerBinding(unitId)
						return true
					end
				end

				task.wait(RETRY_DELAY)
			end
		end

		return false, "Sell не был принят сервером."
	end

	local function playTarget(action)
		local desiredMode = tostring(action.Mode or "")

		if desiredMode == "" then
			return false, "В Target отсутствует режим."
		end

		for attempt = 1, MAX_RETRIES do
			local tower = findTower(action, action.Level)

			if not tower then
				task.wait(RETRY_DELAY)
			else
				if not towerMatchesAction(tower, action) then
					return false,
						"Привязанный Target-юнит имеет другое имя."
				end

				local config = tower:FindFirstChild("Config")
				local targetLevel =
					tonumber(readValue(config, "Level")) or 1

				if action.Level
					and targetLevel ~= tonumber(action.Level) then
					return false,
						"Уровень юнита не совпадает перед Target."
				end

				local currentMode = tostring(
					readValue(config, "TargetMode") or ""
				)

				if currentMode == desiredMode then
					return true
				end

				pcall(function()
					ChangeTowerMode:InvokeServer(tower)
				end)

				task.wait(0.15)
			end

			if not isGameActive() then
				break
			end
		end

		return false,
			"Не удалось установить Target " .. desiredMode .. "."
	end

	local function getSkipButton()
		local gameGui =
			player.PlayerGui:FindFirstChild("GameGui")

		local skipFrame = gameGui
			and gameGui:FindFirstChild("SkipButton")

		local button = skipFrame
			and skipFrame:FindFirstChild("Skip")

		return button and button:IsA("GuiButton")
			and button
			or nil
	end

	local function playSkip(action)
		if not VoteSkip or not VoteSkip:IsA("RemoteEvent") then
			return false, "RemoteEvent VoteSkip не найден."
		end

		local expectedWave = tonumber(action.Wave) or 0

		if expectedWave > 0 and currentWave() > expectedWave then
			return true
		end

		local startingWave = currentWave()

		for attempt = 1, MAX_RETRIES do
			while isGameActive()
				and currentWave() == startingWave do
				local button = getSkipButton()

				if button
					and button.Visible
					and button.Interactable then
					break
				end

				task.wait(0.1)
			end

			if currentWave() ~= startingWave then
				return true
			end

			pcall(function()
				VoteSkip:FireServer()
			end)

			local confirmationStarted = os.clock()

			while isGameActive()
				and currentWave() == startingWave
				and os.clock() - confirmationStarted < 4 do
				local skipPlayers =
					tonumber(
						readValue(info, "SkipPlayers")
					) or 0

				local needed =
					tonumber(readValue(info, "Needed")) or 0

				if needed > 0 and skipPlayers >= needed then
					return true
				end

				task.wait(0.1)
			end

			if currentWave() ~= startingWave then
				return true
			end

			task.wait(RETRY_DELAY)
		end

		return false, "Skip не был подтверждён."
	end

	local function playSpeed(action)
		if not Speed or not Speed:IsA("RemoteEvent") then
			return false, "RemoteEvent Speed не найден."
		end

		local speed = tonumber(action.Speed)

		if not speed then
			return false, "В Speed отсутствует значение."
		end

		local success = pcall(function()
			Speed:FireServer(speed)
		end)

		return success,
			success and nil or "Speed не был отправлен."
	end

	local function playAction(action)
		if action.Type == "Place" then
			return playPlace(action)
		elseif action.Type == "Upgrade" then
			return playUpgrade(action)
		elseif action.Type == "Sell" then
			return playSell(action)
		elseif action.Type == "Target" then
			return playTarget(action)
		elseif action.Type == "Skip" then
			return playSkip(action)
		elseif action.Type == "Speed" then
			return playSpeed(action)
		end

		return false,
			"Неизвестное действие " .. tostring(action.Type)
	end

	local function handlePostGame()
		task.wait(0.75)

		if STRATEGY.AutoReplay then
			local exitGame = events:FindFirstChild("ExitGame")

			if exitGame and exitGame:IsA("RemoteEvent") then

						.. " пропущено: группа "
						.. tostring(action.Group)
						.. " отключена."

		end

	end

end

local autosaveWriting = false
local autosavePending = false

saveAutosave = function()
	if not recording or typeof(writefile) ~= "function" then
		return false
	end

	if autosaveWriting then
		autosavePending = true
		return true
	end

	autosaveWriting = true

	repeat
		autosavePending = false

		local content = serializeStrategy()
		local success = pcall(function()
			writefile(AUTOSAVE_FILE_NAME, content)
		end)

		if not success then
			autosaveWriting = false
			setDebug(
				"Autosave",
				"Не удалось обновить Strat_Autosave.txt.",
				"error"
			)
			return false
		end

		task.wait()
	until not autosavePending or not recording

	autosaveWriting = false
	updateLinoriaLabel(
		"Autosave",
		"Autosave: Strat_Autosave.txt • "
			.. tostring(#recordedActions)
			.. " действий"
	)
	return true
end

	if typeof(readfile) ~= "function"
		or typeof(writefile) ~= "function"
		or typeof(isfile) ~= "function" then
		setDebug(
			"Autosave",
			"Функции readfile/writefile/isfile недоступны.",
			"error"
		)
		return false
	end

	if not isfile(AUTOSAVE_FILE_NAME) then
		setDebug(
			"Autosave",
			"Strat_Autosave.txt не найден.",
			"waiting"
		)
		return false
	end

	local success, errorMessage = pcall(function()
		writefile(
			STRATEGY_FILE_NAME,
			readfile(AUTOSAVE_FILE_NAME)
		)
	end)

	if not success then
		setDebug(
			"Autosave",
			"Не удалось восстановить запись: "
				.. tostring(errorMessage),
			"error"
		)
		return false
	end

	setDebug(
		"Autosave",
		"Восстановил Strat_Autosave.txt в Workspace > Strat.txt.",
		"success"
	)
	return true
end

local function stopRecording(shouldSave, discardPendingSells)
	if not recording then
		setDebug(
			"Recorder",
			"Запись уже остановлена.",
			"waiting"
		)
		return false
	end

	if discardPendingSells then
		table.clear(recentRemovedTowers)
	elseif flushPendingSells then
		task.wait(0.35)
		flushPendingSells()
	end

	recording = false
	updateRecorderStatus()

	if shouldSave then
		return saveStrategy()
	end

	setDebug(
		"Recorder",
		"Запись остановлена.",
		"success"
	)

	return true
end

local function isUpgradeMatch(previousSnapshot, newSnapshot)
	if newSnapshot.Level > previousSnapshot.Level then
		return true
	end

	if previousSnapshot.NextPrice
		and newSnapshot.Price > 0 then
		return math.abs(
			newSnapshot.Price - previousSnapshot.NextPrice
		) < 0.01
	end

	return false
end

connectTowerTargetRecorder = function(tower)
	if towerTargetConnections[tower] then
		return
	end

	local config = tower:FindFirstChild("Config")
	local targetMode = config
		and config:FindFirstChild("TargetMode")

	if not targetMode or not targetMode:IsA("ValueBase") then
		return
	end

	towerTargetConnections[tower] =
		targetMode.Changed:Connect(function(newMode)
			local snapshot = getTowerSnapshot(tower)

			if snapshot then
				appendTargetAction(
					snapshot,
					tostring(newMode)
				)
			end
		end)
end

local function findUpgradeSource(newSnapshot)
	local deadline = os.clock() + 0.2

	repeat
		local bestIndex = nil
		local bestSnapshot = nil
		local bestDistance = TOWER_MATCH_DISTANCE

		for index = #recentRemovedTowers, 1, -1 do
			local removed = recentRemovedTowers[index]
			local age = os.clock() - removed.RemovedAt

			if age > UPGRADE_MATCH_WINDOW then
				table.remove(recentRemovedTowers, index)
			else
				local distance = (
					removed.Snapshot.Position
						- newSnapshot.Position
				).Magnitude

				if distance <= bestDistance
					and isUpgradeMatch(
						removed.Snapshot,
						newSnapshot
					) then
					bestIndex = index
					bestSnapshot = removed.Snapshot
					bestDistance = distance
				end
			end
		end

		if bestIndex then
			table.remove(recentRemovedTowers, bestIndex)
			return bestSnapshot
		end

		task.wait(0.03)
	until os.clock() >= deadline

	return nil
end

local function connectTowerRecorder()
	local towersFolder = workspace:WaitForChild("Towers", 60)

	if not towersFolder then
		return
	end

	for _, tower in ipairs(towersFolder:GetChildren()) do
		local snapshot = getTowerSnapshot(tower)

		if snapshot then
			knownTowerSnapshots[tower] = snapshot
			connectTowerTargetRecorder(tower)
		end
	end

	towersFolder.ChildRemoved:Connect(function(tower)
		local snapshot = knownTowerSnapshots[tower]
			or getTowerSnapshot(tower)

		knownTowerSnapshots[tower] = nil

		local targetConnection =
			towerTargetConnections[tower]

		if targetConnection then
			targetConnection:Disconnect()
			towerTargetConnections[tower] = nil
		end

		local info = workspace:FindFirstChild("Info")

		if not recording
			or not snapshot
			or getValue(info, "GameRunning") ~= true then
			consumedUpgradeSources[tower] = nil
			return
		end

		if consumedUpgradeSources[tower] then
			consumedUpgradeSources[tower] = nil
			return
		end

		local removedEntry = {
			Snapshot = snapshot,
			RemovedAt = os.clock(),
			Time = roundNumber(
				os.clock() - recordingStartedAt,
				3
			),
			Wave = getCurrentWave(),
		}

		removedEntry.Timing =
			getCurrentTiming(removedEntry.Time)

		table.insert(recentRemovedTowers, removedEntry)

		task.delay(UPGRADE_MATCH_WINDOW + 0.1, function()
			for index, entry in ipairs(recentRemovedTowers) do
				if entry == removedEntry then
					table.remove(recentRemovedTowers, index)

					local info =
						workspace:FindFirstChild("Info")

					if recording
						and getValue(
							info,
							"GameRunning"
						) == true then
						appendSellAction(
							snapshot,
							removedEntry
						)
					end

					break
				end
			end
		end)
	end)

	towersFolder.ChildAdded:Connect(function(tower)
		task.delay(0.12, function()
			local snapshot = getTowerSnapshot(tower, true)

			if not snapshot then
				return
			end

			if not recording then
				snapshot.UnitId = getOrAssignTowerId(tower)
				knownTowerSnapshots[tower] = snapshot
				return
			end

			local previousSnapshot =
				findUpgradeSource(snapshot)

			if previousSnapshot and previousSnapshot.UnitId then
				towerIds[tower] = previousSnapshot.UnitId
				snapshot.UnitId = previousSnapshot.UnitId
			else
				snapshot.UnitId = getOrAssignTowerId(tower)
			end

			knownTowerSnapshots[tower] = snapshot
			connectTowerTargetRecorder(tower)

			if previousSnapshot then
				appendUpgradeAction(
					snapshot,
					previousSnapshot
				)
			else
				appendPlaceAction(snapshot)
			end
		end)
	end)
end

local function updateUnitInspector()
	local tower = selectedInspectorTower

	if not tower or not tower.Parent then
		clearInspectorSelection()
		return
	end

	local snapshot = getTowerSnapshot(tower)

	if not snapshot then
		clearInspectorSelection()
		return
	end

	local totalSpent = 0
	local linkedAction = nil
	local sortedActions = table.clone(recordedActions)

	table.sort(sortedActions, function(a, b)
		return (tonumber(a.Time) or 0)
			< (tonumber(b.Time) or 0)
	end)

	for index, action in ipairs(sortedActions) do
		if tonumber(action.UnitId) == tonumber(snapshot.UnitId) then
			if action.Type == "Place"
				or action.Type == "Upgrade" then
				totalSpent = totalSpent
					+ (tonumber(action.Price) or 0)
			end

			linkedAction = index
		end
	end

	updateLinoriaLabel(
		"InspectorName",
		"Юнит: " .. snapshot.Name
	)
	updateLinoriaLabel(
		"InspectorId",
		"ID: #" .. tostring(snapshot.UnitId or "—")
	)
	updateLinoriaLabel(
		"InspectorLevel",
		"Уровень: " .. tostring(snapshot.Level)
	)
	updateLinoriaLabel(
		"InspectorTarget",
		"Target Mode: " .. tostring(snapshot.TargetMode)
	)
	updateLinoriaLabel(
		"InspectorPosition",
		"Координаты: " .. formatCoordinates(snapshot.Position)
	)
	updateLinoriaLabel(
		"InspectorSpent",
		"Потрачено: " .. tostring(totalSpent)
	)
	updateLinoriaLabel(
		"InspectorAction",
		"Строка Strat.txt: "
			.. (linkedAction and ("#" .. linkedAction) or "ещё не записан")
	)
end

			.. " #"

	)
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
	if gameProcessed then
		return
	end

	if input.UserInputType ~= Enum.UserInputType.MouseButton1
		and input.UserInputType ~= Enum.UserInputType.Touch then
		return
	end

	local camera = workspace.CurrentCamera
	local towersFolder = workspace:FindFirstChild("Towers")

	if not camera or not towersFolder then
		return
	end

	local ray = camera:ViewportPointToRay(
		input.Position.X,
		input.Position.Y
	)
	local parameters = RaycastParams.new()
	parameters.FilterType = Enum.RaycastFilterType.Exclude
	parameters.FilterDescendantsInstances = player.Character
		and {player.Character}
		or {}

	local result = workspace:Raycast(
		ray.Origin,
		ray.Direction * 2000,
		parameters
	)

	if not result then
		return
	end

	local current = result.Instance

	while current and current.Parent ~= towersFolder do
		current = current.Parent
	end

	if current and current:IsA("Model") then
		selectInspectorTower(current)
	end
end)

local function getNativeSpeedFromGui()
	local fallbackSpeed = nil

	for _, guiName in ipairs({"Speed2x", "Speed3x"}) do
		local speedGui = playerGui:FindFirstChild(guiName)
		local frame = speedGui
			and speedGui:FindFirstChild("Frame")

		local status = frame
			and frame:FindFirstChild("Status")

		if status
			and (status:IsA("TextLabel")
				or status:IsA("TextButton")) then
			local speed = string.match(status.Text, "x(%d+)")

			if speed then
				local numericSpeed = tonumber(speed)

				if speedGui:IsA("ScreenGui")
					and speedGui.Enabled then
					return numericSpeed
				end

				fallbackSpeed = fallbackSpeed or numericSpeed
			end
		end
	end

	return fallbackSpeed
end

local function updateNativeSpeedGui(speed)
	for _, guiName in ipairs({"Speed2x", "Speed3x"}) do
		local speedGui = playerGui:FindFirstChild(guiName)
		local frame = speedGui
			and speedGui:FindFirstChild("Frame")

		local status = frame
			and frame:FindFirstChild("Status")

		if status
			and (status:IsA("TextLabel")
				or status:IsA("TextButton")) then
			status.Text = "x" .. tostring(speed) .. " Speed"
		end
	end
end

requestGameSpeed = function(speed, forceRequest)
	speed = tonumber(speed)

	if not speed or not isGameActive() then
		return false
	end

	if not forceRequest and currentGameSpeed == speed then
		return true
	end

	local now = os.clock()

	if not forceRequest
		and now - lastSpeedRequestAt < 1.5 then
		return false
	end

	local events = ReplicatedStorage:FindFirstChild("Events")
	local speedEvent = events and events:FindFirstChild("Speed")

	if not speedEvent or not speedEvent:IsA("RemoteEvent") then
		return false
	end

	lastSpeedRequestAt = now

	local success = pcall(function()
		speedEvent:FireServer(speed)
	end)

	return success
end

requestX2Speed = function()
	if not autoX2SpeedEnabled then
		return false
	end

	return requestGameSpeed(2, false)
end

local function getSkipButton()
	local gameGui = playerGui:FindFirstChild("GameGui")
	local skipFrame = gameGui
		and gameGui:FindFirstChild("SkipButton")

	local skipButton = skipFrame
		and skipFrame:FindFirstChild("Skip")

	if skipButton and skipButton:IsA("GuiButton") then
		return skipButton
	end

	return nil
end

local function tryAutoSkip()
	if not autoSkipEnabled or not isGameActive() then
		return
	end

	local waveNumber = getCurrentWave()

	if waveNumber <= 0
		or lastAutoSkipWave == waveNumber then
		return
	end

	local skipButton = getSkipButton()

	if not skipButton
		or not skipButton.Visible
		or not skipButton.Interactable then
		return
	end

	local events = ReplicatedStorage:FindFirstChild("Events")
	local voteSkip = events and events:FindFirstChild("VoteSkip")

	if not voteSkip or not voteSkip:IsA("RemoteEvent") then
		return
	end

	lastAutoSkipWave = waveNumber
	skipButton.Interactable = false
	skipButton.BackgroundColor3 =
		Color3.fromRGB(110, 110, 110)

	local success = pcall(function()
		voteSkip:FireServer()
	end)

	if not success then
		lastAutoSkipWave = nil
	end
end

local function checkConfirmedSkip(info)
	if not recording then
		return
	end

	local skipPlayers =
		tonumber(getValue(info, "SkipPlayers")) or 0

	local needed =
		tonumber(getValue(info, "Needed")) or 0

	local waveNumber =
		tonumber(getValue(info, "Wave")) or 0

	if needed > 0
		and skipPlayers >= needed
		and waveNumber > 0 then
		appendSkipAction(waveNumber)
	end
end

local function connectMatchAutomation()
	local info = workspace:WaitForChild("Info", 60)
	local events = ReplicatedStorage:WaitForChild("Events", 60)

	if not info or not events then
		return
	end

	currentGameSpeed = getNativeSpeedFromGui()
		or currentGameSpeed

	local speedEvent = events:FindFirstChild("Speed")

	if speedEvent and speedEvent:IsA("RemoteEvent") then
		speedEvent.OnClientEvent:Connect(function(_, speed)
			local confirmedSpeed = tonumber(speed)

			if not confirmedSpeed then
				return
			end

			currentGameSpeed = confirmedSpeed
			updateNativeSpeedGui(confirmedSpeed)
			appendSpeedAction(confirmedSpeed)

			if autoX2SpeedEnabled
				and confirmedSpeed ~= 2 then
				task.delay(0.35, requestX2Speed)
			end
		end)
	end

	local skipPlayers = info:FindFirstChild("SkipPlayers")
	local wave = info:FindFirstChild("Wave")

	if skipPlayers and skipPlayers:IsA("ValueBase") then
		local previousSkipCount =
			tonumber(skipPlayers.Value) or 0

		skipPlayers.Changed:Connect(function()
			local newSkipCount =
				tonumber(skipPlayers.Value) or 0

			if newSkipCount > previousSkipCount then
				checkConfirmedSkip(info)
			end

			previousSkipCount = newSkipCount
		end)
	end

	if wave and wave:IsA("ValueBase") then
		wave.Changed:Connect(function()
			if autoX2SpeedEnabled then
				task.defer(requestX2Speed)
			end
		end)
	end

	while screenGui.Parent do
		tryAutoSkip()

		if autoX2SpeedEnabled then
			requestX2Speed()
		end

		task.wait(0.25)
	end
end

local function handleGameEnd()
	if gameEndHandled then
		return
	end

	gameEndHandled = true

	if recording then
		stopRecording(true, true)
	end

	task.delay(0.75, function()
		local events = ReplicatedStorage:FindFirstChild("Events")

		if not events then
			return
		end

		if autoReplayEnabled then
			local exitGame = events:FindFirstChild("ExitGame")

			if exitGame and exitGame:IsA("RemoteEvent") then
				exitGame:FireServer()
			end
		elseif autoBackToLobbyEnabled then
			local lobbyTeleport =
				events:FindFirstChild("LobbyTeleport")

			if lobbyTeleport
				and lobbyTeleport:IsA("RemoteEvent") then
				lobbyTeleport:FireServer()
			end
		end
	end)
end

local syncingLinoria = false

	local toggle = linoriaToggles[toggleName]

local function setAutoReplay(value)
	autoReplayEnabled = value == true

	if autoReplayEnabled then
		autoBackToLobbyEnabled = false
	end

	profileSnapshot.Settings.AutoReplay = autoReplayEnabled
	profileSnapshot.Settings.AutoBackToLobby =
		autoBackToLobbyEnabled

	renderMiscSettings()
	saveSnapshot()
	syncLinoriaToggle("AutoReplay", autoReplayEnabled)
	syncLinoriaToggle("AutoBack", autoBackToLobbyEnabled)
end

local function setAutoBack(value)
	autoBackToLobbyEnabled = value == true

	if autoBackToLobbyEnabled then
		autoReplayEnabled = false
	end

	profileSnapshot.Settings.AutoReplay = autoReplayEnabled
	profileSnapshot.Settings.AutoBackToLobby =
		autoBackToLobbyEnabled

	renderMiscSettings()
	saveSnapshot()
	syncLinoriaToggle("AutoReplay", autoReplayEnabled)
	syncLinoriaToggle("AutoBack", autoBackToLobbyEnabled)
end

local function setAutoSkip(value)
	autoSkipEnabled = value == true
	profileSnapshot.Settings.AutoSkip = autoSkipEnabled

	if autoSkipEnabled then
		lastAutoSkipWave = nil
		task.defer(tryAutoSkip)
	end

	renderMiscSettings()
	saveSnapshot()
	syncLinoriaToggle("AutoSkip", autoSkipEnabled)
end

local function setAutoX2Speed(value)
	autoX2SpeedEnabled = value == true
	profileSnapshot.Settings.AutoX2Speed =
		autoX2SpeedEnabled

	renderMiscSettings()
	saveSnapshot()
	syncLinoriaToggle("AutoX2", autoX2SpeedEnabled)

	if autoX2SpeedEnabled then
		currentGameSpeed = getNativeSpeedFromGui()
			or currentGameSpeed

		task.defer(requestX2Speed)
	else
		requestGameSpeed(1, true)
	end
end

local function loadLinoriaModule(relativePath)
	if typeof(loadstring) ~= "function" then
		error("loadstring недоступен")
	end

	local source = game:HttpGet(
		LINORIA_REPOSITORY .. relativePath
	)
	local chunk, compileError = loadstring(source)

	if not chunk then
		error(compileError or ("Не удалось загрузить " .. relativePath))
	end

	return chunk()
end

		)

task.spawn(function()
	local eventsObject = ReplicatedStorage:WaitForChild("Events")
	local voteForMapObject = eventsObject:WaitForChild("VoteForMap")

	if not voteForMapObject:IsA("RemoteEvent") then
		setDebug(
			"Ошибка",
			"VoteForMap не является RemoteEvent.",
			"error"
		)
		return
	end

	local updateVoteCountObject = eventsObject:FindFirstChild(
		"UpdateVoteCount"
	)

	if not updateVoteCountObject then
		updateVoteCountObject = eventsObject:WaitForChild(
			"UpdateVoteCount",
			30
		)
	end

	local voteConfirmed = false

	local function isInterfaceVisible(object)
		local current = object

		while current do
			if current:IsA("GuiObject")
				and not current.Visible then
				return false
			end

			if current:IsA("ScreenGui")
				and not current.Enabled then
				return false
			end

			current = current.Parent
		end

		return true
	end

	local function getBossRushButton()
		local gameGui = playerGui:FindFirstChild("GameGui")

		if not gameGui then
			return nil
		end

		local voteFrame = gameGui:FindFirstChild(
			"VoteFrame",
			true
		)

		if not voteFrame then
			return nil
		end

		local bossRushMode = voteFrame:FindFirstChild(
			"BossRushMode",
			true
		)

		if not bossRushMode then
			return nil
		end

		local voteFrameContainer =
			bossRushMode:FindFirstChild(
				"Vote Frame",
				true
			)

		if not voteFrameContainer then
			return nil
		end

		local voteButton =
			voteFrameContainer:FindFirstChild(
				"Vote",
				true
			)

		if voteButton and voteButton:IsA("GuiButton") then
			return voteButton
		end

		return nil
	end

	local function hasOurBossRushVote(voteInformation)
		if typeof(voteInformation) ~= "table" then
			return false
		end

		local bossRushVotes =
			voteInformation[VOTE_MODE_NAME]

		if typeof(bossRushVotes) ~= "table" then
			return false
		end

		for _, userId in pairs(bossRushVotes) do
			if tonumber(userId) == player.UserId then
				return true
			end
		end

		return false
	end

	if updateVoteCountObject
		and updateVoteCountObject:IsA("RemoteEvent") then
		updateVoteCountObject.OnClientEvent:Connect(
			function(voteInformation)
				if hasOurBossRushVote(voteInformation) then
					voteConfirmed = true

					setDebug(

						"success"
					)

		)

		"Ожидание",
		"Жду появление голосования Boss Rush...",
		"waiting"
	)

	while not voteButton do
		local foundButton = getBossRushButton()

		if foundButton
			and isInterfaceVisible(foundButton) then
			voteButton = foundButton
			break
		end

		task.wait(0.25)
	end

		"Голосование",
		"Boss Rush найден. Отправляю голос..."
	)

	voteForMapObject:FireServer(VOTE_MODE_NAME)

		and os.clock() - confirmationStarted
			< VOTE_CONFIRMATION_TIMEOUT do

			"Повторная попытка",
			"Повторно голосую за Boss Rush..."

		task.wait(1)

			setDebug(
				"Голос отправлен",
				"Запрос Boss Rush отправлен серверу.",
				"success"
			)

local function getCharacter()
	local character =
		player.Character or player.CharacterAdded:Wait()

	character:WaitForChild("HumanoidRootPart")
	return character
end

local function getElevatorMapName(elevator)
	local mapLabel = elevator:FindFirstChild(
		"mapName",
		true
	)

	if mapLabel and mapLabel:IsA("TextLabel") then
		return mapLabel.Text
	end

	local titleAttribute =
		elevator:GetAttribute("Title")

	if typeof(titleAttribute) == "string" then
		return titleAttribute
	end

	return nil
end

local function getElevatorPlayerCount(elevator)
	local attributeValue =
		elevator:GetAttribute("Players")

	if typeof(attributeValue) == "number" then
		return attributeValue
	end

	local statusGui = elevator:FindFirstChild(
		"StatusGui",
		true
	)

	local titleLabel =
		statusGui and statusGui:FindFirstChild("Title")

	if titleLabel and titleLabel:IsA("TextLabel") then
		local currentPlayers = string.match(
			titleLabel.Text,
			"^(%d+)%s*/"
		)

		if currentPlayers then
			return tonumber(currentPlayers) or 0
		end
	end

	return 0
end

local function getElevatorId(elevator)
	local config = elevator:FindFirstChild("Config")
	local idValue =
		config and config:FindFirstChild("idk")

	if idValue and idValue:IsA("StringValue") then
		return idValue.Value
	end

	return elevator.Name
end

local function getElevatorFloor(elevator)
	local base = elevator:FindFirstChild("Base", true)

	if base and base:IsA("BasePart") then
		return base
	end

	for _, descendant in ipairs(elevator:GetDescendants()) do
		if descendant:IsA("BasePart")
			and descendant.CanCollide then
			return descendant
		end
	end

	return nil
end

local function isToiletHQElevator(elevator)
	return elevator:IsA("Model")
		and getElevatorMapName(elevator) == TARGET_MAP
		and getElevatorFloor(elevator) ~= nil
end

local function findFreeElevator(ignoredElevators)
	local character = getCharacter()
	local root =
		character:FindFirstChild("HumanoidRootPart")

	if not root or not root:IsA("BasePart") then
		return nil
	end

	local candidates = {}

	for _, elevatorObject in ipairs(
		elevatorsFolder:GetChildren()
	) do
		if ignoredElevators[elevatorObject] then
			continue
		end

		if not elevatorObject:IsA("Model") then
			continue
		end

		if not isToiletHQElevator(elevatorObject) then
			continue
		end

		local currentPlayers =
			getElevatorPlayerCount(elevatorObject)

		if currentPlayers
			>= START_IF_PLAYERS_BELOW then
			continue
		end

		local floor = getElevatorFloor(elevatorObject)

		if not floor then
			continue
		end

		table.insert(candidates, {
			Elevator = elevatorObject,
			Players = currentPlayers,
			Distance = (
				root.Position - floor.Position
			).Magnitude,
		})
	end

	table.sort(candidates, function(a, b)
		if a.Players ~= b.Players then
			return a.Players < b.Players
		end

		return a.Distance < b.Distance
	end)

	if candidates[1] then
		return candidates[1].Elevator
	end

	return nil
end

local function moveIntoElevator(elevator)
	local character = getCharacter()
	local floor = getElevatorFloor(elevator)

	if not floor then
		error(
			"Не найдена деталь пола в "
				.. elevator:GetFullName()
		)
	end

	setDebug(
		"Перемещение",
		"Перемещаю игрока в " .. elevator.Name .. "..."
	)

	local height =
		floor.Size.Y / 2 + TELEPORT_HEIGHT

	character:PivotTo(
		floor.CFrame * CFrame.new(0, height, 0)
	)
end

local function waitForElevatorEntry(
	elevator,
	playersBeforeEntry
)
	local startedAt = os.clock()

	setDebug(
		"Проверка входа",
		"Ожидаю подтверждение входа в лифт...",
		"waiting"
	)

	repeat
		local currentPlayers =
			getElevatorPlayerCount(elevator)

		if currentPlayers > playersBeforeEntry then
			return currentPlayers
		end

		task.wait(0.1)
	until os.clock() - startedAt >= ENTER_TIMEOUT

	return getElevatorPlayerCount(elevator)
end

local function startAutoToiletHQ()
	local ignoredElevators = {}

	while true do
		setDebug(
			"Поиск лифта",
			"Ищу свободный лифт Toilet HQ...",
			"waiting"
		)

		local elevator =
			findFreeElevator(ignoredElevators)

		if not elevator then
			setDebug(
				"Лифт не найден",
				"Свободный Toilet HQ не найден.",
				"error"
			)
			return false
		end

		ignoredElevators[elevator] = true

		local elevatorId =
			getElevatorId(elevator)

		local playersBeforeEntry =
			getElevatorPlayerCount(elevator)

		setDebug(
			"Лифт найден",
			elevator.Name
				.. " • игроков: "
				.. tostring(playersBeforeEntry)
		)

		moveIntoElevator(elevator)
		task.wait(0.25)

		setDebug(
			"Вход в лифт",
			"Отправляю EnterElevator..."
		)

		enterElevatorEvent:FireServer(elevatorId)

		local currentPlayers =
			waitForElevatorEntry(
				elevator,
				playersBeforeEntry
			)

		task.wait(0.25)

		currentPlayers =
			getElevatorPlayerCount(elevator)

		if currentPlayers
			< START_IF_PLAYERS_BELOW then

			setDebug(
				"Сохранение",
				"Сохраняю колоду, монеты и гемы..."
			)

			captureDeck()
			captureLobbyCurrencies()
			renderProfile()
			saveSnapshot()

			task.wait(0.2)

			setDebug(
				"Запуск",
				"Запускаю Toilet HQ • игроков: "
					.. tostring(currentPlayers)
			)

			startElevatorEvent:FireServer(elevatorId)

			setDebug(
				"Лифт запущен",
				"Данные сохранены. Жду Boss Rush...",
				"success"
			)

			return true
		end

		setDebug(
			"Лифт занят",
			"В лифте уже "
				.. tostring(currentPlayers)
				.. " игроков. Ищу другой...",
			"waiting"
		)

		leaveElevatorEvent:FireServer(elevatorId)
		task.wait(0.5)
	end
end

local function startLocalStrategyModule()
	if not AUTO_START_LOCAL_STRATEGY then
		return false
	end

	local strategyModule = script:FindFirstChild(
		LOCAL_STRATEGY_MODULE_NAME
	)

	if not strategyModule then
		setDebug(
			"Strategy",
			"Добавь дочерний ModuleScript Strat для автозапуска.",
			"waiting"
		)
		return false
	end

	if not strategyModule:IsA("ModuleScript") then
		setDebug(
			"Strategy",
			"Объект Strat должен быть ModuleScript.",
			"error"
		)
		return false
	end

	task.defer(function()
		local loaded, result = pcall(require, strategyModule)

		if not loaded then
			setDebug(
				"Strategy error",
				tostring(result),
				"error"
			)
			warn("[HAMSTER HUB] Не удалось запустить Strat:", result)
			return
		end

		setDebug(
			"Strategy",
			"Strat загружен. Replay запущен автоматически.",
			"success"
		)
	end)

	return true
end
